cat > /etc/nginx/conf.d/cnxs-mgmt-bitbucket.conf << 'EOF'
server {
    server_name bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov;
    listen 80;
    return 301 https://$http_host$request_uri;
}

server {
    listen 443 ssl http2;
    server_name bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov;

    ssl_certificate     /etc/nginx/certificates/mgmt_wildcard.pem;
    ssl_certificate_key /etc/nginx/certificates/privatekey.key;

    resolver 127.0.0.11 ipv6=off;

    location / {
        proxy_pass http://cnxs-mgmt-bitbucket:7990;

        proxy_set_header Host              $host;
        proxy_set_header X-Forwarded-Host  $host;
        proxy_set_header X-Forwarded-Port  $server_port;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-Server $host;

        proxy_http_version 1.1;
        proxy_read_timeout 600s;
        proxy_send_timeout 600s;
    }
}
EOF
