    @Test
    public void verifyClinIcbSREChargeCodeIsNotAllowedToChangeAfterSOCN() {
        SOCNRequest.Body body1 = socnRequest.getBody().get(0);
        body1.setChargingFrequencyAndSreElementCode("USAGE");
        SOCNRequest.Body body2 = socnRequest.getBody().get(1);
        body2.setChargingFrequencyAndSreElementCode("02");
        List<UBIDetail> ubiDetails = SOCNMockHelper.buildUBIDetails(socnRequest);
        body1.setContractLineItemNumber("AA0001");
        body1.setIndividualCaseBasisCodeNumber("AA0001Icb");
        body1.setChargingFrequencyAndSreElementCode("MRC");
        body2.setContractLineItemNumber("AA0002");
        body2.setIndividualCaseBasisCodeNumber("AA0002Icb");
        body2.setChargingFrequencyAndSreElementCode("03");
        List<String> ubis = socnRequest.getBody().stream().map(SOCNRequest.Body::getUniqueBillingIdentifier).collect(Collectors.toList());
        mockUBIDetails(ubis, ubiDetails);
        when(socnValidatorHandler.isValidFirmOrderCommitmentDate(any(), any())).thenReturn(true);

        socnDirectValidatorBean.validateDirectSocnRequest(socnRequest, null, internalAccount, orderCaches.getContractorOrderCache(),
                                                          dataTransactionSequenceNumber, AGENCY_HIERARCHY_CODES);

        SOCNConexusResponse socnConexusResponse = (SOCNConexusResponse) messageContext.get(SOCN_RESPONSE_KEY);
        List<ResponseType> responseTypes = socnConexusResponse.getResponses().stream()
                                                              .filter(response -> (response.getCode().longValue() == WebServiceMessages.CODE_20016.getCode()))
                                                              .collect(Collectors.toList());
        assertThat(responseTypes.size(), is(4));
        List<String> expectedMessages = new ArrayList<>();
        String[] fields = {"Contract Line Item Number", "Individual Case Basis Code Number", "Charging Frequency and SRE Element Code"};
        for (SOCNRequest.Body socnBody : socnRequest.getBody()) {
            for (String field : fields) {
                if (SOCNMockHelper.SRE_PRICING_ELEMENT_CODES.contains(socnBody.getChargingFrequencyAndSreElementCode())) {
                    expectedMessages.add(socnBody.getUniqueBillingIdentifier() + ":" + socnBody.getChargingFrequencyAndSreElementCode() + " " + WebServiceMessages.CODE_20016.getMessage(field, field));
                } else {
                    expectedMessages.add(socnBody.getUniqueBillingIdentifier() + " " + WebServiceMessages.CODE_20016.getMessage(field, field));
                }
            }
        }

        for (ResponseType responseType : responseTypes) {
            assertTrue(expectedMessages.contains(responseType.getMessage()));
        }

        reset(socnValidatorHandler, cachePopulator);
    }

    @Test
    public void testGetClosestMatchingInventory() {
        SOCNRequest.Body body1 = socnRequest.getBody().get(0);
        body1.setChargingFrequencyAndSreElementCode("USAGE");
        SOCNRequest.Body body2 = socnRequest.getBody().get(1);
        body1.setContractLineItemNumber("AA0001");
        body1.setIndividualCaseBasisCodeNumber("AA0001Icb");
        body1.setChargingFrequencyAndSreElementCode("MRC");
        body2.setContractLineItemNumber("EQ90001");
        body2.setIndividualCaseBasisCodeNumber("EQIcb");
        body2.setChargingFrequencyAndSreElementCode("03");
        List<UBIDetail> ubiDetails = SOCNMockHelper.buildUBIDetails(socnRequest);
        ubiDetails.add(SOCNMockHelper.buildUBIDetail(taskOrder, body1.getUniqueBillingIdentifier(), body1.getContractLineItemNumber(), "icb", "NRC", StringUtils.randomAlphaNumeric(5)));

        UBIDetail ubiDetail = socnDirectValidatorBean.getClosestMatchingUBI(ubiDetails, body1);

        assertThat(ubiDetail.getClin(), is(body1.getContractLineItemNumber()));
        assertThat(ubiDetail.getCaseNumber(), is(body1.getIndividualCaseBasisCodeNumber()));
        assertThat(ubiDetail.getChargingFrequencyType().getCode(), is(body1.getChargingFrequencyAndSreElementCode()));

        body1.setChargingFrequencyAndSreElementCode("USAGE");
        ubiDetail = socnDirectValidatorBean.getClosestMatchingUBI(ubiDetails, body1);
        assertThat(ubiDetail.getClin(), is(body1.getContractLineItemNumber()));
        assertThat(ubiDetail.getCaseNumber(), is(body1.getIndividualCaseBasisCodeNumber()));
    }

    @Test
    public void testPopulateOrderItemCollectionsWithAllPossibleOrderItemTypeBodies() {
        Set<String> allUbis = new HashSet<>();
        Set<String> emails = new HashSet<>();
        Set<String> countryNames = new HashSet<>();
        Set<Integer> jurisdictionIds = new HashSet<>();
        Set<String> networkSiteCodes = new HashSet<>();
        Set<String> agencyHierarchyCodes = new HashSet<>();
        Set<String> clins = new HashSet<>();
        OrderItemType[] orderItemTypes = OrderItemType.values();
        socnRequest = SOCNRequestFactory.INSTANCE.createCompleteSOCNRequest(orderItemTypes.length);
        setLowerCaseData(socnRequest.getHeader());
        for (int i = 0; i < orderItemTypes.length; i++) {
            SOCNRequest.Body socnBody = socnRequest.getBody().get(i);
            socnBody.setOrderItemTypeCode(orderItemTypes[i].getCode());
            if (i == 0) {
                socnBody.setSitePocEmailAddress(null);
                socnBody.setAltSitePocEmailAddress(null);
                socnBody.setLocationState(null);
                socnBody.setLocationProvince(null);
                socnBody.setTerminatingIconectivNsc(null);
                socnBody.setLocationLatitude(BigDecimal.valueOf(12345.12345678));
                socnBody.setLocationLongitude(BigDecimal.valueOf(21345.12345678));
            } else {
                socnBody.setSitePocEmailAddress(i + "@email.com");
                socnBody.setAltSitePocEmailAddress(i + "@alt.email.com");
                socnBody.setLocationState("sTate" + i);
                socnBody.setLocationProvince("pRoVince" + i);
                socnBody.setTerminatingIconectivNsc("TNSC-" + i);
            }
            socnBody.setLocationCountry("country" + i);
            socnBody.setIconectivNsc("NSC-" + i);
            socnBody.setAgencyHierarchyCode("AHC-" + i);
        }
        socnDirectValidatorBean.populateOrderItemCollections(socnRequest, allUbis, ubiSet, emails, countryNames, jurisdictionIds,
                                                             networkSiteCodes, agencyHierarchyCodes, clins);

        assertThat(allUbis.size(), is(orderItemTypes.length));
        assertThat(ubiSet.size(), is(orderItemTypes.length - 1));
        assertThat(emails.size(), is(((orderItemTypes.length - 1) * 2) + 2));
        assertThat(emails, hasItems("1@EMAIL.COM", "1@ALT.EMAIL.COM", socnRequest.getHeader().getContractingOfficerRepresentativeEmailAddress().toUpperCase(),
                                    socnRequest.getHeader().getTechnicalPointOfContactEmailAddress().toUpperCase()));
        assertThat(countryNames.size(), is(orderItemTypes.length));
        assertThat(jurisdictionIds.size(), is(1));
        assertThat(networkSiteCodes.size(), is((orderItemTypes.length * 2) - 1));
        assertThat(agencyHierarchyCodes.size(), is(orderItemTypes.length));
        verifyCaseChange();
        verifyScaleChange();
    }

    private void setLowerCaseData(SOCNRequest.Header socnHeader) {
        socnHeader.setOrderHeaderTypeCode(socnHeader.getOrderHeaderTypeCode().toLowerCase());
        socnHeader.setOrderMethodCode(socnHeader.getOrderMethodCode().toLowerCase());
    }

    private void verifyScaleChange() {
        assertThat(socnRequest.getBody().get(0).getLocationLatitude(), is(BigDecimal.valueOf(12345.123457)));
        assertThat(socnRequest.getBody().get(0).getLocationLongitude(), is(BigDecimal.valueOf(21345.123457)));
    }

    private void verifyCaseChange() {
        assertThat(socnRequest.getHeader().getOrderHeaderTypeCode(), is(socnRequest.getHeader().getOrderHeaderTypeCode().toUpperCase()));
        assertThat(socnRequest.getHeader().getOrderMethodCode(), is(socnRequest.getHeader().getOrderMethodCode().toUpperCase()));
        for (int i = 0; i < socnRequest.getBody().size(); i++) {
            SOCNRequest.Body socnBody = socnRequest.getBody().get(i);
            assertThat(socnBody.getLocationCountry(), is("COUNTRY" + i));
        }
    }

    @Test
    public void verifyTerminatingNSCRequiredForCertainPricePaths() throws IllegalAccessException {
        List<MasterClin> masterClins = new ArrayList<>();
        MasterClin clinWithPricePath01_OWS = SOCNMockHelper.mockMasterClin("OWS1234", "OWS", PricePath.ONE, ChargingFrequencyType.MRC);
        masterClins.add(clinWithPricePath01_OWS);
        masterClins.add(SOCNMockHelper.mockMasterClin("OWS1234-2", "OWS", PricePath.ONE, ChargingFrequencyType.NRC));
        MasterClin masterClinWithNullRule = SOCNMockHelper.mockMasterClin("OWS1234-1", "OWS", PricePath.ONE, ChargingFrequencyType.NRC);
        when(masterClinWithNullRule.getRule()).thenReturn(null);
        masterClins.add(masterClinWithNullRule);
        masterClins.add(SOCNMockHelper.mockMasterClin("OWS1234-3", "OWS", PricePath.ONE, ChargingFrequencyType.USAGE));
        masterClins.add(SOCNMockHelper.mockMasterClin("ME1234", "Metro", PricePath.ONE, ChargingFrequencyType.MRC));
        masterClins.add(SOCNMockHelper.mockMasterClin("PL00108", "PLS", PricePath.ONE, ChargingFrequencyType.NRC));
        MasterClin clinWithPricePath01_Port = SOCNMockHelper.mockMasterClin("VN00004", "PORT", PricePath.ONE, ChargingFrequencyType.MRC);
        masterClins.add(clinWithPricePath01_Port);
        masterClins.add(SOCNMockHelper.mockMasterClin("EN00023", "EVC", PricePath.FOUR, ChargingFrequencyType.MRC));
        masterClins.add(SOCNMockHelper.mockMasterClin("EN00023-1", "EVC", PricePath.FOUR, ChargingFrequencyType.USAGE));
        masterClins.add(SOCNMockHelper.mockMasterClin("PL00108", "PLS", PricePath.EIGHT, ChargingFrequencyType.MRC));
        masterClins.add(SOCNMockHelper.mockMasterClin("PL00108-1", "PLS", PricePath.EIGHT, ChargingFrequencyType.USAGE));
        MasterClin clinWithPricePath01 = SOCNMockHelper.mockMasterClin("OW00101", "Metro", PricePath.TEN, ChargingFrequencyType.MRC);
        masterClins.add(clinWithPricePath01);
        masterClins.add(SOCNMockHelper.mockMasterClin("OW00101-2", "Metro", PricePath.TEN, ChargingFrequencyType.NRC));
        masterClins.add(SOCNMockHelper.mockMasterClin("OW00101-3", "Metro", PricePath.TEN, ChargingFrequencyType.USAGE));
        SOCNRequest socnRequest = SOCNRequestFactory.INSTANCE.createCompleteSOCNRequest(masterClins.size());
        List<String> expectedMessages = new ArrayList<>();
        ClinDetailsCache clinDetailsCache = new ClinDetailsCache();
        clinDetailsCache.populateCache(masterClins);
        String terminatingNscField = "Terminating iconectiv NSC";
        for (int i = 0; i < socnRequest.getBody().size(); i++) {
            SOCNRequest.Body socnBody = socnRequest.getBody().get(i);
            MasterClin masterClin = masterClins.get(i);
            socnBody.setContractLineItemNumber(masterClin.getClin());
            InventoryKey inventoryKey = new InventoryKey(socnBody.getUniqueBillingIdentifier(), socnRequest.getHeader().getContractNumber(), socnBody.getChargingFrequencyAndSreElementCode());
            if (!masterClin.getClin().equals(clinWithPricePath01_OWS.getClin())
                    && !masterClin.getClin().equals(clinWithPricePath01.getClin())) {
                socnBody.setTerminatingIconectivNsc(null);
                if (masterClin.getChargingFrequencyType() != ChargingFrequencyType.USAGE
                        && !masterClin.getClin().equals(clinWithPricePath01_Port.getClin())) {
                    expectedMessages.add(WebServiceMessages.CODE_20003.getMessage(getMessagePrefix(inventoryKey) + " " + terminatingNscField));
                }
            }
        }
        assertThat(expectedMessages, hasSize(7));

        FieldUtils.writeField(orderCaches, "clinDetailsCache", clinDetailsCache, true);
        socnDirectValidatorBean.validateDirectSocnRequest(socnRequest, taskOrder, internalAccount, orderCaches.getContractorOrderCache(),
                                                          dataTransactionSequenceNumber, AGENCY_HIERARCHY_CODES);

        //verify
        SOCNConexusResponse socnConexusResponse = (SOCNConexusResponse) messageContext.get(SOCN_RESPONSE_KEY);
        List<ResponseType> responseTypes = socnConexusResponse.getResponses()
                                                              .stream()
                                                              .filter(response -> (response.getCode().longValue() == WebServiceMessages.CODE_20003.getCode()
                                                                      && response.getMessage().contains(WebServiceMessages.CODE_20003.getMessage(terminatingNscField))))
                                                              .collect(Collectors.toList());
        assertThat(responseTypes.size(), is(expectedMessages.size() - 1));
        for (ResponseType responseType : responseTypes) {
            assertTrue(expectedMessages.contains(responseType.getMessage()));
        }
        reset(socnValidatorHandler, cachePopulator);
    }

    protected InventoryKey mockChangeOrderBodyItems(QuantityChangeEffectiveDateCache quantityChangeEffectiveDateCache,

    private void runSOCNDateValidation(SOCNRequest.Body socnBody, OrderType orderType, OrderItemType orderItemType, boolean errorExpected) {
        LocalDate socnDate = LocalDate.now().minusDays(1);
        socnRequest.getHeader().setServiceOrderCompletionDate(socnDate);
        socnRequest.getHeader().setOrderHeaderTypeCode(orderType.getCode());
        socnBody.setOrderItemTypeCode(orderItemType.getCode());
        ubiDetail = null;
        //Run 1
        getSOCNCentralAndDirectValidatorBean().validateSOCNDateIsNotBeforeInstallDateForChangeAndDisconnect(socnRequest, socnBody, ubiDetail, inventoryKeyData);
        //verify
        SOCNTestHelper.INSTANCE.assertSocnResponseWithNoError(messageContext);
        //Run 2
        ubiDetail = mock(UBIDetail.class);
        getSOCNCentralAndDirectValidatorBean().validateSOCNDateIsNotBeforeInstallDateForChangeAndDisconnect(socnRequest, socnBody, ubiDetail, inventoryKeyData);
        //verify
        SOCNTestHelper.INSTANCE.assertSocnResponseWithNoError(messageContext);

        LocalDate installDate = LocalDate.now();
        when(ubiDetail.getInstallDate()).thenReturn(installDate);
        socnRequest.getBody().get(0).setOrderItemTypeCode(orderItemType.getCode());
        //run 3
        getSOCNCentralAndDirectValidatorBean().validateSOCNDateIsNotBeforeInstallDateForChangeAndDisconnect(socnRequest, socnBody, ubiDetail, inventoryKeyData);
        //verify
        if (errorExpected) {
            SOCNTestHelper.INSTANCE.assertSocnResponseWithResponses(messageContext, LevelType.ERROR,
                                                                    getMessagePrefix(inventoryKeyData) + " Service Order Completion Date cannot precede Install Date", WebServiceMessages.CODE_20015);
        } else {
            SOCNTestHelper.INSTANCE.assertSocnResponseWithNoError(messageContext);
        }
        messageContext.put(SOCN_RESPONSE_KEY, SOCNTestHelper.INSTANCE.createSocnResponse(socnRequest));
    }
