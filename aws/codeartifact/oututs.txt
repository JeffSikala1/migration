select u.username, u.realm,
       string_agg(g.group_name, ', ' order by g.group_name) as groups
from access_users u
left join access_users_groups ug on ug.user_id = u.user_id
left join access_groups g on g.group_id = ug.group_id
group by u.user_id, u.username, u.realm
order by regexp_replace(u.username, '@.*$', ''), u.username;

select u.realm as user_realm, ug.realm as membership_realm, count(*)
from access_users_groups ug join access_users u using (user_id)
group by 1, 2 order by 1, 2;

select group_name, realm, realm_attributes from access_groups order by group_name;
