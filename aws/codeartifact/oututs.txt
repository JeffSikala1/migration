#!/opt/rh/rh-python36/root/bin/python3

import datetime
import pandas as pd
import email, smtplib
import sys
import os.path
from os import path
from email.mime.base import MIMEBase
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

today = datetime.date.today()
file_exists = path.exists('/app/certificates_exp_checking_scripts/certificates/EXPIRING_90_60_30.csv')
#print(file_exists)

if file_exists:
  df = pd.read_csv('/app/certificates_exp_checking_scripts/certificates/EXPIRING_90_60_30.csv', sep=',')
else:
  sys.exit()

payload = "Hello,<br><br>This is a friendly notice the certificate(s) listed below are <b>EXPIRING</b> within 90 days.<br><br>"
payload += df.to_html(index=False, justify='left')
payload += "<br><br>Kind Regards,<br><br>The Certificate Check Bot"

print (payload)

subject = 'Vendor Certificate Notification (EXPIRING IN 90 DAYS OR LESS) - ' + str(today.strftime('%m/%d/%Y'))
sender_email = "root@gsa.gov"

receiver_email = ["conexus_devsecops@gsa.gov","jennifer.bowman@gsa.gov","peter.cox@gsa.gov","conexus.system@gsa.gov"]

for i in receiver_email:
  # Create a multipart message and set headers
  message = MIMEMultipart()
  message["From"] = sender_email
  message["To"] = i
  message["Subject"] = subject
  #message["Bcc"] = receiver_email  # Recommended for mass emails

  # Add body to email
  message.attach(MIMEText(payload, "html"))
  text = message.as_string()
  s = smtplib.SMTP('localhost')
  s.sendmail(sender_email, i, text)
  s.quit()
