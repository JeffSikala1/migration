#!/usr/bin/env bash

git_exec=/usr/bin/git

# Parse input
parse_arguments() {
 salt_api_password=""
 salt_api_username="salt-api"
 start_or_finish="start"

 while getopts ":u:p:m:" opt; do
  case $opt in
    u)
      salt_api_username="${OPTARG}" >&2
      ;;
    p)
      salt_api_password="${OPTARG}" >&2
      ;;
    m)
      start_or_finish="${OPTARG}" >&2
      ;;
    l)
      if [ "${OPTARG}" == "debug" ]; then
       set -x
      elif [ "${OPTARG}" == "verbose" ]; then
       set -v
      fi
      ;;
    \?)
      echo "Invalid option: -${OPTARG}" >&2
      exit 1
      ;;
    :)
      echo "Option -${OPTARG} requires an argument." >&2
      exit 1
      ;;
  esac
 done
}

# Exit with a non zero status and an optional message
# $1 - Error message
exit_error()
{
 [ -n "${1}" ] && echo "ERROR: ${1}"
 exit 1
}

# Ensure that we do not have a zero length value
# $1 - Value to test
# $2 - Error message if validation fails
validate_not_empty()
{
 [ -z "${1}" ] && exit_error "${2}"
}

# Get the docker.oracle value off of develop
# $1 - Release version
develop_docker_oracle()
{
 _SUP_=$(${git_exec} checkout develop)
 grep -oP '(?<=<docker.oracle>)[^<]*' pom.xml
 _SUP=$(${git_exec} checkout release/${1})
}

# Login to the salt api and return the token
# $1 - username
# $2 - password
get_salt_token()
{
 curl -k https://conexus-saltmaster.edc.ds1.usda.gov:8000/login \
    -H 'Accept: application/json' \
    -d username="${1}" \
    -d password="${2}" \
    -d eauth=pam \
    2>/dev/null \
    | grep -oP '"token": "\K[^\"]*'
}

# Tag a docker image on the remote build servers
# $1 - Src image
# $2 - Dst Image
# $3 - Salt Api Token
tag_docker_image()
{
 echo "Tagging ${1} as ${2}"
 curl -k https://conexus-saltmaster.edc.ds1.usda.gov:8000/ \
     -H "Accept: application/x-yaml" \
     -H "X-Auth-Token: ${3}"\
     -d client=local \
     -d tgt="dots.prod.oireasd01[29,30]*" \
     -d fun=dockerng.tag \
     -d arg="${1}" \
     -d arg="${2}" \
 || exit_error "Failed to tag docker oracle image"
}

# Update docker.oracle in the env pom
# $1 - Value to find and replace
# $2 - Value to replace found with
update_docker_in_pom()
{
 echo "Updating docker.oracle in Env pom ..."
 cp pom.xml{,.orig}
 # Use parameter expansion to escape possible backslashes in $1 or $2
 sed -i "s/${1//\//\\/}/${2//\//\\/}/" pom.xml
 diff pom.xml.orig pom.xml
 rm pom.xml.orig
}

# Commit and push pom changes in env to release branch
# $1 - Version
commit_and_push()
{
 ${git_exec} commit pom.xml -m "Updated docker image to ${1}" \
 || exit_error "Failed to commit updates to pom.xml"
 ${git_exec} push \
 || exit_error "Failed to push updated pom.xml to remote"
}

# Perform prerequsites here
# $1 - "start" or "finish"
pre()
{
 if [ "${1}" == "start" ]; then
  release_branch_version=$(git branch | grep -oP '(?<=\* release/).*')
  validate_not_empty "${salt_api_password}" "Salt API password not provided"
  # Get the api token to be used when tagging
  token=$(get_salt_token ${salt_api_username} ${salt_api_password})
  validate_not_empty "${token}" "Failed to get salt-api token"
 else
  ${git_exec} remote set-url origin ssh://git@oireasd0120.edc.ds1.usda.gov:7999/cxn/env.git
  ${git_exec} pull
  release_branch_version=$(git branch -r | grep -oP '(?<=release/).*')
  develop_oracle_image=$(develop_docker_oracle "${release_branch_version}")
  validate_not_empty "${develop_oracle_image}" "Failed to get develop docker oracle image"
 fi
 # Get the value of the current docker.oracle on our branch
 current_docker_oracle_image=$(grep -oP '(?<=<docker.oracle>)[^<]*' pom.xml)

 # Perform a quick sanity check before moving on
 validate_not_empty "${release_branch_version}" "Unable to set release branch version.  Bailing out"
 validate_not_empty "${current_docker_oracle_image}" "Failed to find current oracle image"

 [ "${1}" == "finish" ] && ( git checkout release/${release_branch_version} \
 || exit_error "Failed to checkout release/${release_branch_version}" )
}

# Perform release start functions
start_release()
{
 # Tag the oracle image for the release
 tag_docker_image \
  ${current_docker_oracle_image} \
  conexus-artifactory.edc.ds1.usda.gov:5000/usda-oracle-12c:${release_branch_version} \
  ${token}
 # Update the env pom
 update_docker_in_pom \
  "\(<docker.oracle>conexus-artifactory.edc.ds1.usda.gov:5000/[^:]\+\)[^<]*\(.*\)" \
  "\1:${release_branch_version}\2"
 # Commit and push
 commit_and_push "${release_branch_version}"
}

# Perform release finish functions
finish_release()
{
# Update the pom.xml back to the conexus-dl tag
 update_docker_in_pom \
  "${current_docker_oracle_image}" "${develop_oracle_image}"
# Push change back to remote
 commit_and_push "conexus-dl"
}

parse_arguments "$@"
pre "${start_or_finish}"
if [ "${start_or_finish}" == "start" ]; then
 start_release
else
 finish_release
fi
