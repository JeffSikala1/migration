# Add the datasource
# Note: The only Conexus component that requires an XA datasource is the reconciliation module since JBoss
# BRMS requires a XA datasource (Ref: https://access.redhat.com/documentation/en-us/red_hat_jboss_brms/6.4/html/installation_guide/chap_special_setups)
# Because of this, I've standardized on XA.

# Why the log allocation-retry numbers?: From Simon: "We initially had low values for these and we increased
# them because we were running into issue when DISC was performing maintenance (patching, networking, etc)
# that would cause the DB to not be available.  Jboss would keep trying for 10 seconds, and then give up and give up on the DB connection.
# Increasing the retry timeout allowed the application to recover on its own."
xa-data-source add \
  --name=XAConexusDS \
  --jndi-name=java:jboss/datasources/XAConexusDS \
  --driver-name=postgres \
  --user-name=${env.CONEXUS_DB_USER:nsfs_app} \
  --password=${env.CONEXUS_DB_PASSWORD:Gray+Gh0st+7} \
  --validate-on-match=true \
  --background-validation=false \
  --valid-connection-checker-class-name=org.jboss.jca.adapters.jdbc.extensions.postgres.PostgreSQLValidConnectionChecker \
  --exception-sorter-class-name=org.jboss.jca.adapters.jdbc.extensions.postgres.PostgreSQLExceptionSorter \
  --same-rm-override=false \
  --no-tx-separate-pool=true \

#!/bin/bash

JBOSS_PATCH_ZIP=$1
# For debugging
set -x

function wait_for_server_start() {
  until `$JBOSS_HOME/bin/jboss-cli.sh -c ":read-attribute(name=server-state)" 2> /dev/null | grep -q "running"`; do
    sleep 1
  done
}
function wait_for_server_stop() {
  until `$JBOSS_HOME/bin/jboss-cli.sh -c ":read-attribute(name=server-state)" 2> /dev/null | grep -q "Could not connect"`; do
    sleep 1
  done
}
function stopJBoss() {
  echo "---------------------------------- Stop Jboss -------------------------------------------------------"
  $JBOSS_HOME/bin/jboss-cli.sh --connect --command=:shutdown
  wait_for_server_stop
}
function startJBoss() {
  echo "---------------------------------- Start JBOSS ---------------------------------------------------------"
  $JBOSS_HOME/bin/standalone.sh -c standalone-full.xml &
  wait_for_server_start
}
function patch() {
  startJBoss

  echo "---------------------------------- Patch Jboss ---------------------------------------------------------"
  echo "y" | $JBOSS_HOME/bin/jboss-cli.sh -c "installer update --maven-repo-files=/app/jboss-eap-patch.zip"
  $JBOSS_HOME/bin/jboss-cli.sh -c "shutdown --perform-installation"
  echo "---------------------------------- Patch applied -------------------------------------------------------"

  wait_for_server_start

  stopJBoss
}
function addDriver() {
  startJBoss
  echo "---------------------------------- Add Database Driver  -----------------------------------------------"
  $JBOSS_HOME/bin/jboss-cli.sh -c --command="module add --name=org.postgres --resources=/tmp/postgresql-42.7.8.jar --dependencies=javax.api,javax.transaction.api"
  stopJBoss
}

function addUser() {
  echo "---------------------------------- Add User  ------------------------------------------------------------"
  # This application realm (-a) user is used by the IT tests to remote to the JBoss instance. See the common-test-utils
  # gov.gsa.nsfs.tests.JndiFixture class.
  # I think the "guest" role is important because I think it is referenced in the JMS security configuration
  # (see the messaging <subsystem>)
  $JBOSS_HOME/bin/add-user.sh -a -u 'nsfsapp' -p '!nsfsapp1' -g 'guest'
  # The user also has to be a management user for domain test ear's StartUpIT
  $JBOSS_HOME/bin/add-user.sh -u 'nsfsapp' -p '!nsfsapp1' -g 'guest'

  # Add user for Rest interface HTTP Basic Auth
  $JBOSS_HOME/bin/add-user.sh -a -g "api" -u "apiuser" -p "PAssword1234$" -r "ApplicationRealm" -e
}

function applyConfiguration() {
  echo "---------------------------------- Apply Jboss Configuration ------------------------------------------"

  $JBOSS_HOME/bin/jboss-cli.sh --file=security.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=propertiesConfig.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=loggingConfig.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=datasourceConfig.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=mailConfig.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=messagingConfig.cli
  $JBOSS_HOME/bin/jboss-cli.sh --file=miscConfig.cli

  echo "---------------------------------- Jboss Configuration Applied ---------------------------------------"
}

if [ -f "$JBOSS_PATCH_ZIP" ]; then patch; fi

addDriver
addUser
applyConfiguration
