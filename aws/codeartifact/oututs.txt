mkdir -p "$HOST_CACHE_BASE"

if ! docker network inspect "$NETWORK_NAME" >/dev/null 2>&1; then
  echo "Creating docker network $NETWORK_NAME ($NETWORK_SUBNET)..."
  docker network create -d bridge --subnet "$NETWORK_SUBNET" "$NETWORK_NAME" >/dev/null
else
  echo "Docker network $NETWORK_NAME already exists."
fi

HOSTNAME_SHORT="$(hostname)"

echo
echo "Stopping/removing existing build agents..."
docker ps -aq --filter "name=${HOSTNAME_SHORT}-ba-" | xargs -r docker rm -f

echo
echo "Starting build agents..."

for num in $(seq 1 "$NUMBER_OF_BUILD_AGENTS"); do
  AGENT_NAME="${HOSTNAME_SHORT}-ba-${num}"
  HOST_AGENT_HOME="${HOST_CACHE_BASE}/${AGENT_NAME}"
  CONTAINER_AGENT_HOME="/cache/bamboo/${AGENT_NAME}"

  mkdir -p "$HOST_AGENT_HOME"
  echo "Agent name: $AGENT_NAME; Host Agent Home: $HOST_AGENT_HOME; Container Agent Home: $CONTAINER_AGENT_HOME; "
  
  #    exec java -Dbamboo.home=${CONTAINER_AGENT_HOME} -jar /opt/bamboo-agent.jar ${BAMBOO_URL}
  docker run -d \
    -v "${SSH_KEYS_DIR}:/opt/bamboo-keys:ro" \
    -v "${HOST_AGENT_HOME}:${CONTAINER_AGENT_HOME}" \
    -v "${DOCKER_SOCK}:${DOCKER_SOCK}" \
    -h "$AGENT_NAME" \
    --net="$NETWORK_NAME" --name="$AGENT_NAME" --init --restart=unless-stopped --sysctl net.ipv4.ip_forward=1 \
    "$DOCKER_BUILD_AGENT_IMAGE" \
    bash -lc "
      set -e
      echo 'startup umask before change:' \$(umask)
      umask 0002
      echo 'startup umask after change:' \$(umask)
      mkdir -p /opt/bamboo-keys
      mkdir -p ${CONTAINER_AGENT_HOME}
      exec java -Dbamboo.home=${CONTAINER_AGENT_HOME} -jar /opt/bamboo-agent.jar ${BAMBOO_URL}
    "
done
#      mkdir -p /opt/java
#      ln -sfn /etc/alternatives/java_sdk_21 /opt/java/openjdk

echo
echo "Running Bamboo agent containers:"
sleep 5
docker ps --filter "name=${HOSTNAME_SHORT}-ba-" --format "table {{.Names}}\t{{.Image}}\t{{.Status}}"
