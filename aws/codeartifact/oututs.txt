        # Copy in the SSH Key

        curl -k -u sa_conexus:${bamboo.artifactory_auth_secret} "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/keys/s_bamboo_git" -o ~/.ssh/id_rsa
        chmod 600 ~/.ssh
        chmod 400 ~/.ssh/id_rsa

        # Configure git to use this key


 
      description: Install SSH key
  - any-task:
      plugin-key: com.atlassian.bamboo.plugins.variable.updater.variable-updater-generic:variable-extractor
      configuration:
        variable: currentVersion
        removeSnapshot: 'true'
        variableScope: JOB
      description: Get current version
  - script:
      interpreter: SHELL
      scripts:
      - |-
        #!/usr/bin/env bash
        echo "Checking for tag version ${bamboo_currentVersion}"

        git remote set-url origin ${bamboo.planRepository.repositoryUrl}

        # If the currentVersionm, from the previous Maven version variable
        # extractor, is also a tag version, then the tag version/name will be in
        # the variable $does_tag_exist
        # Else $does_tag_exist will be a null string
        does_tag_exist=$(git tag -l ${bamboo.currentVersion})
        echo $does_tag_exist

        if [ $does_tag_exist ]; then
          echo "Deleting the tag $does_tag_exist"
          echo "git tag -d $does_tag_exist"
          git tag -d $does_tag_exist
          echo "git push --delete origin $does_tag_exist"
          git push --delete origin $does_tag_exist
        else
            echo "No tag found"
        fi
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes'
      description: Delete tag, if it already exists
  - script:
      interpreter: SHELL
      scripts:
      - "#!/usr/bin/env bash\n\ngit remote set-url origin ${bamboo.planRepository.repositoryUrl}\ngit pull\nrelease_branch=$(git branch -r | grep -oP '(?<=origin\\/)release\\/.*')\n\n# Handle the case of more than one release branch\nif [ -n \"${release_branch}\" ]; then\n  for i in \"${release_branch[@]}\"; do\n    echo \"Deleting the release branch $i\"\n        git push origin --delete $i\n        git branch -D $i\n        git fetch\n  done\nelse\n        echo \"No release branch\"\n        exit 0 \nfi"
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes'
      description: Delete Release Branch
  - script:
      interpreter: SHELL
      scripts:
      - |-
        #!/usr/bin/env bash

        git checkout develop
        git branch -a
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes'
      description: Pull Latest Develop branch
  - maven:
      executable: Maven 3
      jdk: JDK 21
      goal: -U -B external.atlassian.jgitflow:jgitflow-maven-plugin:release-start -DkeepBranch=true -DpushReleases=true -DallowSnapshots=true -DenableSshAgent=true -DdefaultOriginUrl=${bamboo.planRepository.repositoryUrl} -s /usr/share/maven/conf/settings-release.xml
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes' ARTIFACTORY_UPLOADER_SECRET="${bamboo.ARTIFACTORY_UPLOADER_SECRET}"
      description: GitFlow Release Start
  - script:

        #git push --set-upstream origin $BRANCH_NAME

      description: Update SNAPSHOT with RC
