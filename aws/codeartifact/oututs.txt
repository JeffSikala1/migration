# Install to /etc/systemd/system/artifactory.service on ip-10-56-112-243
#   sudo mkdir -p /app/artifactory-startup
#   sudo cp start-artifactory.sh lib-common.sh /app/artifactory-startup/
#   sudo chmod +x /app/artifactory-startup/start-artifactory.sh
#   sudo cp artifactory.service /etc/systemd/system/

# NOT touch bamboo.service or bitbucket.service, and does not restart prod-postgres if it's
# already running.

# the other host's units.

[Service]
Type=oneshot
RemainAfterExit=yes
WorkingDirectory=/app/artifactory-startup
ExecStart=/app/artifactory-startup/start-artifactory.sh
TimeoutStartSec=180
