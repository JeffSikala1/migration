  - script:
      interpreter: SHELL
      scripts:
      - |-
        #!/usr/bin/env bash
        echo "Checking for tag version ${bamboo_currentVersion}"

        git remote set-url origin ${bamboo.planRepository.repositoryUrl}

        # If the currentVersionm, from the previous Maven version variable
        # extractor, is also a tag version, then the tag version/name will be in
        # the variable $does_tag_exist
        # Else $does_tag_exist will be a null string
        does_tag_exist=$(git tag -l ${bamboo.currentVersion})
        echo $does_tag_exist

        if [ $does_tag_exist ]; then
          echo "Deleting the tag $does_tag_exist"
          echo "git tag -d $does_tag_exist"
          git tag -d $does_tag_exist
          echo "git push --delete origin $does_tag_exist"
          git push --delete origin $does_tag_exist
        else
            echo "No tag found"
        fi
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes'
      description: Delete tag, if it already exists

      interpreter: SHELL
      scripts:
      - |-
        #!/usr/bin/env bash

        git remote set-url origin ${bamboo.planRepository.repositoryUrl}
        git pull

        release_branch=$(git branch -r | grep -oP '(?&lt;=origin\/)release\/.*')

        echo "Release branch is set from grep above is $release_branch"

        if [ -z "${release_branch}" ]; then
            echo "Failed to match release branch from the following: "
            git branch -r
            exit 1
        else
            echo "Checking out ${release_branch}"
            git checkout "${release_branch}"
        fi
      environment: GIT_SSH_COMMAND='ssh -i ~/.ssh/id_rsa -o IdentitiesOnly=yes'
      description: Checkout release branch
