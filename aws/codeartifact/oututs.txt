[sikaj01@ip-10-56-112-33 apache]$ cat ../../deploy/updatemanifest.sh
#!/usr/bin/env bash

# Debugging
set -x
# Quit on any error
set -e



usage() { echo "Usage: $0 -s <uipublic|portal|jms|webservice|brms|reporting> " 1>&2; exit 1; }

while getopts ":s:" opt; do
    case "${opt}" in
        s)
            s=${OPTARG}
            ((s == uipublic || s == portal || s == jms || s == webservice || s == brms || s == reporting )) || usage
            ;;
        :)
            echo "Error: Missing argument for -${OPTARG} "
            usage
            ;;
        *)
            usage
            ;;
    esac
done
shift $((OPTIND-1))

if [ -z "${s}" ] ; then
    usage
fi

echo "s = ${s}"
module_group="${s}"


echo "bamboo_deploy_version = $bamboo_deploy_version"
echo "bamboo_deploy_release = $bamboo_deploy_release"
echo "bamboo_deploy_project = $bamboo_deploy_project"

PROGDIR=$(dirname ${BASH_SOURCE})
echo "${PROGDIR}"
cd ${PROGDIR}
k8sfolder="../k8s"

cd $k8sfolder

e="${bamboo_deploy_environment}"
source ${k8sfolder}/jboss${e}.bashenv    

case "$s" in
    uipublic)
        export uipublic_image="${bamboo_deploy_release}"
        envsubst '$uipublic_image'  <${k8sfolder}/apache/run-httpd.yaml > ${k8sfolder}/apache/${e}/run-httpd.yaml
        ;;
    brms)
        export brms_image="${bamboo_deploy_release}"
        #sed -i -E "s~(brms_image=).*$~\1\"${val}\"~" $k8sfolder/jboss${e}.bashenv
        envsubst '$brms_image $brms_pod_mem $brms_pod_mem_limit $brms_jboss_xmx'  <${k8sfolder}/jboss/run-brms.yaml > ${k8sfolder}/jboss/${e}/run-brms.yaml
        ;;
    jms)
        export jms_image="${bamboo_deploy_release}"
        envsubst '$jms_image $jms_pod_mem $jms_pod_mem_limit $jms_jboss_xmx'  <${k8sfolder}/jboss/run-jms.yaml > ${k8sfolder}/jboss/${e}/run-jms.yaml
        ;;
    portal)
        export portal_image="${bamboo_deploy_release}"
        envsubst '$portal_image $portal_pod_mem $portal_pod_mem_limit $portal_jboss_xmx'  <${k8sfolder}/jboss/run-portal.yaml > ${k8sfolder}/jboss/${e}/run-portal.yaml
        ;;
    webservice)
        export webservice_image="${bamboo_deploy_release}"
        envsubst '$webservice_image $webservice_pod_mem $webservice_pod_mem_limit $webservice_jboss_xmx'  <${k8sfolder}/jboss/run-ws.yaml > ${k8sfolder}/jboss/${e}/run-ws.yaml
        ;;
    reporting)
        export reporting_image="${bamboo_deploy_release}"
        envsubst '$reporting_image $reporting_pod_mem $reporting_pod_mem_limit $reporting_jboss_xmx'  <${k8sfolder}/jboss/run-reporting.yaml > ${k8sfolder}/jboss/${e}/run-reporting.yaml
        ;;
    *)
        echo "Failed to get jboss module type ${s} "
        ;;
esac
