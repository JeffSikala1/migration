bamboo=> SELECT b.build_id, b.full_key, b.title, b.build_type
FROM build b
WHERE (b.full_key LIKE 'BT-%' OR b.full_key LIKE 'SR-%' OR b.full_key LIKE 'FR-%')
  AND b.build_type = 'CHAIN'
  AND b.marked_for_deletion = false
  AND b.build_id NOT IN (SELECT plan_id FROM plan_vcs_specs_source)
ORDER BY b.full_key;
