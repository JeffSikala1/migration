tail -5 /opt/jfrog/artifactory/var/log/artifactory-request.log


docker exec artifactory bash -c '
for repo in docker-local docker-wso2mi npm-local pypi-local rpm-local yum-local docker-remote bintray-docker-remote npm-remote centos rpm-remote nodesource12x nodesource8x; do
  live=$(grep -c "/artifactory/${repo}/" /opt/jfrog/artifactory/var/log/artifactory-request.log 2>/dev/null || echo 0)
  archived=$(zgrep -c "/artifactory/${repo}/" /opt/jfrog/artifactory/var/log/archived/artifactory-request.*.log.gz 2>/dev/null | awk -F: "{sum+=\$2} END {print sum+0}")
  echo "$repo: live=$live archived=$archived"
done
'
