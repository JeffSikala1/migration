for repo in docker-local docker-wso2mi npm-local pypi-local rpm-local yum-local; do
  echo "=== $repo ==="
  curl -s -H "Authorization: Bearer $artifactory_admin_api_key_secret" \
    -H "Content-Type: text/plain" \
    -X POST "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/search/aql" \
    -d "items.find({\"repo\":\"$repo\"}).include(\"name\",\"stat.downloads\",\"stat.downloaded\")" \
    | jq '[.results[] | select(.stat.downloads != null and .stat.downloads > 0)] | {items_with_downloads: length, most_recent: (map(.stat.downloaded) | max)}'
done
