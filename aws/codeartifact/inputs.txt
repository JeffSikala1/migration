DELETE FROM plan_vcs_specs_source
WHERE plan_id IN (
  SELECT b.build_id
  FROM build b
  WHERE (b.full_key LIKE 'BT-%' OR b.full_key LIKE 'SR-%')
    AND b.build_type = 'CHAIN'
    AND b.marked_for_deletion = false
);

    AND b.build_type = 'CHAIN'
    AND b.marked_for_deletion = false
);
