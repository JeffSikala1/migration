BEGIN;

-- Count what we're about to delete
SELECT COUNT(*) AS will_delete
FROM plan_vcs_specs_source pvss
JOIN build b ON b.build_id = pvss.plan_id
WHERE b.build_type = 'CHAIN'
  AND b.marked_for_deletion = false;

-- Delete spec links for all CHAIN plans
DELETE FROM plan_vcs_specs_source
WHERE plan_id IN (
  SELECT b.build_id
  FROM build b
  WHERE b.build_type = 'CHAIN'
    AND b.marked_for_deletion = false
);

-- Confirm rows remaining (should be 0 if all CHAINs were detached;
-- if there are still rows, they're for JOB-level builds, not CHAINs)
SELECT COUNT(*) AS remaining_links
FROM plan_vcs_specs_source;

-- If counts look right: COMMIT;
-- If not:               ROLLBACK;
