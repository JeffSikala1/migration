create_user() {
  u="$1"; uid="$2"; key="$3"
  id "$u" >/dev/null 2>&1 || useradd -m -u "$uid" -g users -G wheel "$u"
  id -nG "$u" | grep -qw wheel || usermod -aG wheel "$u"
  install -d -m 700 -o "$u" -g users "/home/$u/.ssh"
  grep -qxF "$key" "/home/$u/.ssh/authorized_keys" 2>/dev/null || echo "$key" >> "/home/$u/.ssh/authorized_keys"
  chown "$u:users" "/home/$u/.ssh/authorized_keys"; chmod 600 "/home/$u/.ssh/authorized_keys"
  restorecon -R "/home/$u/.ssh"
}
