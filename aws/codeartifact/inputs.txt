# Memory per uid at the 02:30 OOM event (rss is in 4K pages)
journalctl -b -1 -k --since "2026-10-23 02:30:00" --until "2026-10-23 02:45:00" --no-pager \
 | sed -E 's/.*kernel: +//; s/[][]/ /g' \
 | awk '$1 ~ /^[0-9]+$/ && NF>=9 {rss[$2]+=$5; n[$2]++} END {for (u in rss) printf "%8.0f MB %5d procs uid %s\n", rss[u]*4/1024, n[u], u}' \
 | sort -rn | head -10
