export GIT_SSH_COMMAND='ssh -i /opt/bamboo-keys/bitbucket_ci -o IdentitiesOnly=yes -o StrictHostKeyChecking=accept-new'

git fetch --prune origin

# Check what release branches currently exist
git ls-remote --heads origin 'refs/heads/release/*'

# Delete the four stale branches
for branch in release/01.00.000.107 release/01.00.000.108 release/01.00.000.109 release/01.00.000.110; do
  echo "Deleting ${branch}"
  git push origin --delete "${branch}" || echo "WARNING: could not delete ${branch}"
done

# Confirm release/01.00.000.111 still exists and has RC content
git fetch origin refs/heads/release/01.00.000.111
git show origin/release/01.00.000.111:pom.xml | grep version
