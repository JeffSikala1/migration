docker exec -i dev-postgres psql -U artifactory -d artifactory -c "
SELECT p.name, t.resource_type, t.target_name, a.actions, u.username, g.group_name
FROM access_permissions_v2 p
JOIN access_permissions_targets_v2 t ON t.perm_id = p.perm_id
JOIN access_permissions_actions_v2 a ON a.perm_id = p.perm_id AND a.resource_type = t.resource_type
LEFT JOIN access_users u ON u.user_id = a.user_id
LEFT JOIN access_groups g ON g.group_id = a.group_id
WHERE p.name = 'conexus-deploy';"
