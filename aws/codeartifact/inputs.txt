sudo tee /app/bbmirror/bin/oneshot_verify.sh >/dev/null <<'EOF'
#!/usr/bin/env bash
# Read-only: compare branch+tag refs of every in-scope repo, prod vs dev.
set -uo pipefail
source /app/bbmirror/etc/oneshot.env
EXCLUDE_PROJECTS="${EXCLUDE_PROJECTS:-CNXS}"
api() { curl -sS --fail --max-time 60 -K - -H 'Accept: application/json' "$1$3" <<<"header = \"Authorization: Bearer $2\""; }
refs() { GIT_TERMINAL_PROMPT=0 GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=http.extraHeader GIT_CONFIG_VALUE_0="Authorization: Bearer $2" git ls-remote --heads --tags "$1" 2>&1 | sort; }
start=0; match=0; diff_n=0
while :; do
  r="$(api "$PROD_URL" "$PROD_TOKEN" "/rest/api/1.0/repos?limit=100&start=$start")" || { echo "list failed"; exit 1; }
  while IFS=$'\t' read -r key type slug state; do
    [[ "$type" == PERSONAL ]] && continue
    [[ " ${EXCLUDE_PROJECTS^^} " == *" ${key^^} "* ]] && continue
    lk="${key,,}"
    if diff <(refs "$PROD_URL/scm/$lk/$slug.git" "$PROD_TOKEN") <(refs "$DEV_URL/scm/$lk/$slug.git" "$DEV_TOKEN") >/dev/null; then
      match=$((match+1))
    else
      echo "DIFFER: $key/$slug"; diff_n=$((diff_n+1))
    fi
  done < <(jq -r '.values[] | [.project.key, .project.type, .slug, .state] | @tsv' <<<"$r")
  [[ "$(jq -r .isLastPage <<<"$r")" == true ]] && break
  start="$(jq -r .nextPageStart <<<"$r")"
done
echo "verify: match=$match differ=$diff_n"
EOF
sudo chmod 700 /app/bbmirror/bin/oneshot_verify.sh && sudo bash -n /app/bbmirror/bin/oneshot_verify.sh && echo syntax-ok
