sudo bash -c '
source /app/bbmirror/etc/mirror.env
R=contract-mod
gd=/app/bbmirror/cache/CNXS/$R.git
export GIT_TERMINAL_PROMPT=0 GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=http.extraHeader GIT_CONFIG_VALUE_0="Authorization: Bearer $DEV_TOKEN"
git ls-remote --heads --tags "$DEV_URL/scm/cnxs/$R.git" | grep -v "\^{}" | awk "{print \$2, \$1}" | LC_ALL=C sort -k1,1 > /tmp/dev_$R.txt
git --git-dir=$gd for-each-ref --format="%(refname) %(objectname)" refs/heads refs/tags | LC_ALL=C sort -k1,1 > /tmp/prod_$R.txt
echo "== only in DEV (prune would delete):"
LC_ALL=C join -v1 /tmp/dev_$R.txt /tmp/prod_$R.txt | cut -d" " -f1
echo "== in both, DIFFERENT commit:"
LC_ALL=C join /tmp/dev_$R.txt /tmp/prod_$R.txt | awk "\$2!=\$3 {print \$1}"
echo "== only in PROD (would be created):"
LC_ALL=C join -v2 /tmp/dev_$R.txt /tmp/prod_$R.txt | cut -d" " -f1
'
