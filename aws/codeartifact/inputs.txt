# Which dotfile starts the agent?
sudo grep -n 'ssh-agent' /home/pkundm01/.bashrc /home/pkundm01/.bash_profile /home/pkundm01/.profile /etc/profile.d/* /etc/bashrc 2>/dev/null

# Is it only this account?
ps -eo user=,comm= | awk '$2=="ssh-agent" {n[$1]++} END {for (u in n) print n[u], u}'

# Where are the logins coming from, and how many sessions per user?
sudo journalctl -u sshd --since "-15min" --no-pager | grep -oE 'Accepted publickey for [a-z0-9]+ from [0-9.]+' | sort | uniq -c | sort -rn
loginctl list-sessions --no-legend | awk '{print $3}' | sort | uniq -c
