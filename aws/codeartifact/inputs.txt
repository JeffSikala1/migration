# 1. Clean up stale Bamboo build directories (every 15 min)
#    Removes files unaccessed for 2+ days. Preserves repo caches.

20 */1 * * * /bin/bash /opt/scripts/docker-cleanup.sh >/dev/null 2>&1

*/30 * * * * /bin/bash /opt/scripts/orphaned-test-container-cleanup.sh >/dev/null 2>&1

# 4. Apache Commons Text vulnerability cleanup (every 15 min)
