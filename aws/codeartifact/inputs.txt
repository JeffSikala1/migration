for perm in "conexus-read-only" "Any Remote" "Anything"; do
  echo "=== ${perm} ==="
  curl -sk -u ${ARTIFACTORY_USER}:${ARTIFACTORY_TOKEN} \
    "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/v2/security/permissions/$(echo "${perm}" | sed 's/ /%20/g')" \
    | jq '{name, repo: .repo}'
  echo
done
