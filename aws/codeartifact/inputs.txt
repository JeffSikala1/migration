for repo in env contract-mod common-rest domain reporting ws-client task webservices ws-client ui pricing reconciliation dpa; do
  result=$(git ls-remote --heads \
    ssh://git@bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov:7999/cnxs/${repo}.git \

done
