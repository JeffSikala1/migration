cd /tmp

for repo in common-rest pricing task billing-sftp rest; do
  git clone --bare \
    ssh://git@bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov:7999/cnxs/${repo}.git \
    ${repo}-check.git 2>/dev/null || echo "NOTE: ${repo}-check.git already exists, skipping clone"
done

for repo in common-rest pricing task billing-sftp rest; do
  echo "=== ${repo} ==="
  cd /tmp/${repo}-check.git
  echo "-- develop --"
  git log --oneline -5 develop 2>/dev/null || echo "no develop branch"
  echo "-- main --"
  git log --oneline -5 main 2>/dev/null || echo "no main branch"
  cd /tmp
done
