# ===== S3 layout this produces =====
#
# cnxs-atlassian-backups/
#   artifactory/
#     apps/
#       YYYY-MM-DD/
#         artifactory_var_TIMESTAMP.tar.gz
#         artifactory_security_TIMESTAMP.tar.gz
#     crypto/
#       YYYY-MM-DD/
#         artifactory_keys_TIMESTAMP.tar.gz
#     postgres/
#       YYYY-MM-DD/
#         artifactory_TIMESTAMP.dump
#         globals_TIMESTAMP.sql
#     metadata/
#       YYYY-MM-DD/
#         metadata_TIMESTAMP.json
#     secrets/

#         YYYY-MM-DD/

# Create bucket
aws s3api create-bucket \
  --bucket cnxs-artifactory-backups \
  --region us-east-1

# Block all public access
aws s3api put-public-access-block \
  --bucket cnxs-artifactory-backups \
  --public-access-block-configuration \
    "BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true"

# Enable versioning
aws s3api put-bucket-versioning \
  --bucket cnxs-artifactory-backups \
  --versioning-configuration Status=Enabled

# Match KMS encryption config from cnxs-atlassian-backups
aws s3api put-bucket-encryption \
  --bucket cnxs-artifactory-backups \
  --server-side-encryption-configuration '{
    "Rules": [{
      "ApplyServerSideEncryptionByDefault": {
        "SSEAlgorithm": "aws:kms",
        "KMSMasterKeyID": "arn:aws:kms:us-east-1:339713019047:key/ae797822-83c9-4752-89f1-eb9d3fc53b68"
      },
      "BucketKeyEnabled": true
    }]
  }'

# Lifecycle policy matching Atlassian pattern, adjusted for flat structure
aws s3api put-bucket-lifecycle-configuration \
  --bucket cnxs-artifactory-backups \
  --lifecycle-configuration '{
    "Rules": [
      {
        "ID": "daily retention",
        "Filter": {
          "Prefix": ""
        },
        "Status": "Enabled",
        "Expiration": {
          "Days": 30
        }
      },
      {
        "ID": "weekly retention",
        "Filter": {
          "Prefix": "weekly/"
        },
        "Status": "Enabled",
        "Expiration": {
          "Days": 90
        }
      }
    ]
  }'

# Verify
aws s3api get-bucket-encryption --bucket cnxs-artifactory-backups
aws s3api get-bucket-lifecycle-configuration --bucket cnxs-artifactory-backups
aws s3api get-public-access-block --bucket cnxs-artifactory-backups
