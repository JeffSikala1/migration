CERT=/app/docker/volumes/nginxVolume/_data/certificates/mgmt_wildcard.pem
KEY=/app/docker/volumes/nginxVolume/_data/certificates/privatekey.key
KS=/app/docker/volumes/bambooVolume/_data/shared/configuration/broker.ks
PASS=bamboo

# Backup first
cp -p ${KS} ${KS}.bak.$(date +%F_%H%M%S)

# Convert to PKCS12
openssl pkcs12 -export \
  -in ${CERT} \
  -inkey ${KEY} \
  -out /tmp/broker_new.p12 \
  -name jmsbrokerkey \
  -passout pass:${PASS}



# Delete expired entry
keytool -delete \
  -alias jmsbrokerkey \
  -keystore ${KS} \
  -storepass ${PASS}

# Import new cert from PKCS12
keytool -importkeystore \
  -srckeystore /tmp/broker_new.p12 \
  -srcstoretype PKCS12 \
  -srcstorepass ${PASS} \
  -srcalias jmsbrokerkey \
  -destkeystore ${KS} \
  -deststorepass ${PASS} \
  -destalias jmsbrokerkey \
  -noprompt


keytool -v -list \
  -keystore ${KS} \
  -storepass ${PASS} \
  -alias jmsbrokerkey \
  | grep -E 'Alias|Valid|until|Owner'


docker restart prod-bamboo


docker logs prod-bamboo --follow --since 30s 2>&1 | grep -iE 'jms|broker|ssl|error|started'

rm /tmp/broker_new.p12

docker inspect prod-bamboo | grep -i bamboo.manage.jms.ssl
