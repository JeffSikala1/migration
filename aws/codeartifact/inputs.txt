REPO="ssh://git@bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov:7999/cnxs/ui.git"

# Fetch latest develop
git fetch origin refs/heads/develop

# Create release branch from develop tip

# Convert SNAPSHOT to RC
find . -name pom.xml -exec sed -i 's/-SNAPSHOT/-RC/g' {} +

# Verify
grep -r '01.00.000.111\|SNAPSHOT' pom.xml

# Commit and push
git config user.name "bamboo"
git config user.email "bamboo@local"
git add -A
git commit -m "Update SNAPSHOT to RC"
git push "${REPO}" release/01.00.000.111

# Confirm
