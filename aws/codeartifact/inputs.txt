  --description "Final snapshot before decommission - orphaned pre-migration /mnt/temp volume (old docker data-root + manual root/fstab/atlassian-backups snapshot from before /app/docker migration)" \
  --tag-specifications "ResourceType=snapshot,Tags=[{Key=Name,Value=vol-05ebadfa92a493533-decommission-snapshot},{Key=Purpose,Value=pre-decommission},{Key=Host,Value=ip-10-56-112-34}]"

  --query "Snapshots[0].{State:State,Progress:Progress}"
# poll until State is "completed"






# 4) detach the volume from the instance

# confirm it's fully detached before deleting

  --query "Volumes[0].State"
# poll until "available"

aws ec2 delete-volume --region us-east-1 --volume-id vol-05ebadfa92a493533
