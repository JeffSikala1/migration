for p in $(curl -sf -u admin:${LEGACY_TOKEN} \
  "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions" \
  | jq -r '.[].name'); do

  curl -sf -u admin:${LEGACY_TOKEN} \
    "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions/${p}" \
  | jq -r --arg name "$p" '
      select(.repo.actions.users.anonymous != null)
      | "[\($name)] anonymous=\(.repo.actions.users.anonymous | join(",")) repos=\(.repo.repositories | join(","))"
    '
done
