usage() {
    cat << EOT

  Usage: $SCRIPTNAME [args]

  Examples:

    Build the bamboo build image for cert

    - $SCRIPTNAME -e cert

    Build the bamboo build image for prod

    - $SCRIPTNAME -e prod

    Build the bamboo image version 2020-11-01 for prod

    - $SCRIPTNAME -e prod -v 2020-11-01

    Build the bamboo image with Nodejs 12.x and Angular 8.x 

    - $SCRIPTNAME -a 8 -n 12

  Options:
    -b Centos Base image version (7 or 8) (default is 7)
    -e Environment to build e.g., prod, cert, aws (default prod)
    -c No cache
    -a Angular version (5, 7, 8, etc)  (default is 13)
    -n Nodejs version (12, 14 etc) (default is 16)
    -v Version of the docker image (default is date and time)
    
EOT
}   # ----------  end of function usage  ----------

## If you expect an argument after an option, you need a colon after the letter ##
while getopts ':b:e:ca:n:v:h' opt
do
  case "${opt}" in
    b ) baseImageVersion=${OPTARG}  ;;
    e ) environment=${OPTARG}       ;;
    c ) NOCACHE="1"                 ;;
    a ) angularVersion=${OPTARG}    ;;
    n ) nodeVersion=${OPTARG}       ;;
    v ) dateString=${OPTARG}        ;;
    h ) usage; exit 0               ;;
    \?)  echo
         echo "Option does not exist : ${OPTARG}"
         usage
         exit 1
         ;;
    :)
         echo "Option -${OPTARG} requires an argument." >&2
         exit 1
         ;;
  esac    # --- end of case ---
done
shift $((OPTIND-1))


check_artifactory() {

  fi

#create_dockerfile "$ENV" "$HTTPS" "$NPM" "$RH_USER_NAME" "$RH_PASSWORD"

  local name="$PROD_IMAGE"
elif [ "$environment" = "cert" ]; then
  local name="$CERT_IMAGE"
elif [ "$environment" = "aws" ]; then
  local name="$AWS_IMAGE"
else
  echo "ERROR: incorrect environment name"
  exit 1
fi


if [ $NOCACHE == "1" ]; then
    echo "Building with no-cache"
    /usr/bin/docker build -t "${name}:node${nodeVersion}-angular${angularVersion}-el${baseImageVersion}-${dateString}" --pull --no-cache=true  . || cleanup
else
    echo "Building with cache"
    /usr/bin/docker build -t "${name}:node${nodeVersion}-angular${angularVersion}-el${baseImageVersion}-${dateString}" --pull  . || cleanup
fi
# Export the version to file.properties for Bamboo to use
echo "tag=node${nodeVersion}-angular${angularVersion}-el${baseImageVersion}-${dateString}" > file.properties

} # ----------  end of function create_image  ----------
