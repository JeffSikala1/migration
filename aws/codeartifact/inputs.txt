
# 1) Backup current config
curl -sk -H "Authorization: Bearer $ART_ADMIN_TOKEN" \
  "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/system/configuration" \
  -o /tmp/system-config-backup.xml

# 2) Apply updated config
curl -sk -H "Authorization: Bearer $ART_ADMIN_TOKEN" \
  -H "Content-Type: application/xml" \
  -X PUT \
  --data-binary @/tmp/artifactory-config.xml \
  "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/system/configuration"
