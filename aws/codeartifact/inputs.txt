aws s3api get-bucket-encryption --bucket cnxs-atlassian-backups 2>/dev/null || echo "No bucket-level encryption configured"

aws s3api get-bucket-lifecycle-configuration --bucket cnxs-atlassian-backups 2>/dev/null || echo "No lifecycle policy"



# Create the bucket
aws s3api create-bucket \
  --bucket cnxs-artifactory-backups \
  --region us-east-1

# Block all public access (same posture as cnxs-atlassian-backups)
aws s3api put-public-access-block \
  --bucket cnxs-artifactory-backups \
  --public-access-block-configuration \
    "BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true"

# Enable versioning (good practice for backup buckets)
aws s3api put-bucket-versioning \
  --bucket cnxs-artifactory-backups \
  --versioning-configuration Status=Enabled

# Confirm
aws s3api get-bucket-location --bucket cnxs-artifactory-backups
aws s3api get-public-access-block --bucket cnxs-artifactory-backups
