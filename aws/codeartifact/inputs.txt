for spec in "bamboo:bitbucket[.]mgmt|artifactory[.]mgmt" "bitbucket:bamboo[.]mgmt"; do
  db=${spec%%:*}; pat=${spec#*:}
  echo "== $db"
  sudo docker exec dev-postgres sh -c "pg_dump -U \"\${POSTGRES_USER:-postgres}\" -a $db" \
   | awk -v pat="$pat" '/^COPY /{t=$2} $0 ~ pat {c[t]++} END{for(k in c) print c[k], k}' | sort -rn | head -12
done

grep -rEoh '(ssh|scp|rsync|kubectl|argocd|helm)[^|;&"]{0,80}' bamboo-specs/deployment-specs | sort | uniq -c | sort -rn | head -30
