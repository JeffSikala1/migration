# Check what conexus-plugin-repository (the working virtual) aggregates
curl -sk -u ${ARTIFACTORY_USER}:${ARTIFACTORY_TOKEN} \
  "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/repositories/conexus-plugin-repository" \
  | jq '.repositories'

# Check all permissions sa_bamboo has access to
curl -sk -u ${ARTIFACTORY_USER}:${ARTIFACTORY_TOKEN} \
  "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/v2/security/permissions" \
  | jq '.[].name' \
  | while read name; do
      clean=$(echo "$name" | tr -d '"')
      result=$(curl -sk -u ${ARTIFACTORY_USER}:${ARTIFACTORY_TOKEN} \
        "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/v2/security/permissions/$(echo "${clean}" | sed 's/ /%20/g')" \
        | jq -r '.repo.actions.users.sa_bamboo // empty | join(",")')
      if [ -n "$result" ]; then
        echo "${clean}: ${result}"
      fi
    done



# Remove conexus-ll-plugin-repository from the planned update
jq '.repo.repositories -= ["conexus-ll-plugin-repository"]' \
  /tmp/conexus-read-only-updated.json > /tmp/conexus-read-only-updated2.json
mv /tmp/conexus-read-only-updated2.json /tmp/conexus-read-only-updated.json
