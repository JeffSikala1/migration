# find the existing working Artifactory vhost config to use as the pattern
sudo find / -iname "*.conf" 2>/dev/null | xargs grep -l "artifactory" 2>/dev/null

# confirm nginx container and its config mount
docker inspect dev-nginx --format "{{range .Mounts}}{{.Source}} -> {{.Destination}}{{println}}{{end}}"

# check what cert dev is currently using (need the cert/key filenames to reuse)
docker exec dev-nginx ls -la /etc/nginx/certs/ 2>/dev/null || sudo find / -iname "*.crt" -path "*nginx*" 2>/dev/null
