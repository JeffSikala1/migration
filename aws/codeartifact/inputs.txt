#!/usr/bin/env bash
# vscode-cleanup.sh - show and clean up YOUR VS Code Server processes on the jump box.
# Only touches processes owned by you. Never touches other users or your current shell.
#
# Usage:
#   ./vscode-cleanup.sh            # report only (safe)
#   ./vscode-cleanup.sh --kill     # stop your VS Code server processes
#   ./vscode-cleanup.sh --kill --purge   # also delete old server versions + logs/cache

set -u
ME="$(id -un)"
KILL=0; PURGE=0
for a in "$@"; do
  case "$a" in
    --kill)  KILL=1 ;;
    --purge) PURGE=1 ;;
    *) echo "Unknown option: $a"; exit 1 ;;
  esac
done

# Match only VS Code server pieces owned by me; exclude this script and its grep.
PATTERN='\.vscode-server|vscode-server/cli|server-main\.js|code-[0-9a-f]{10}'
mine() { pgrep -u "$ME" -af "$PATTERN" | grep -v -e "$0" -e pgrep || true; }

echo "== Host load =="
uptime
free -m | awk 'NR==1 || /Mem|Swap/'
echo

echo "== Your VS Code server processes =="
PROCS="$(mine)"
if [ -z "$PROCS" ]; then
  echo "None running."
else
  echo "$PROCS" | cut -c1-160
  echo
  echo "Count: $(echo "$PROCS" | wc -l)"
  echo "Memory (RSS, MB) for your node/code processes:"
  ps -u "$ME" -o rss=,comm= | awk '/node|code/ {s+=$1} END {printf "  %.0f MB\n", s/1024}'
fi
echo

echo "== Defunct (zombie) processes owned by you =="
ps -u "$ME" -o pid=,ppid=,stat=,comm= | awk '$3 ~ /^Z/' || true
echo

echo "== Server installs =="
du -sh "$HOME/.vscode-server" 2>/dev/null || echo "No ~/.vscode-server"
ls -1d "$HOME"/.vscode-server/cli/servers/Stable-* 2>/dev/null || true
echo

if [ "$KILL" -eq 0 ]; then
  echo "Report only. Re-run with --kill to stop these processes."
  exit 0
fi

echo "== Stopping your VS Code server processes =="
PIDS="$(pgrep -u "$ME" -f "$PATTERN" | grep -vx "$$" || true)"
if [ -n "$PIDS" ]; then
  # polite first, then force after 5s
  echo "$PIDS" | xargs -r kill -TERM 2>/dev/null || true
  sleep 5
  LEFT="$(pgrep -u "$ME" -f "$PATTERN" | grep -vx "$$" || true)"
  [ -n "$LEFT" ] && echo "$LEFT" | xargs -r kill -KILL 2>/dev/null || true
fi

# Clean sockets/temp files the server leaves behind
rm -f /tmp/code-* 2>/dev/null || true

if [ "$PURGE" -eq 1 ]; then
  echo "== Purging old server versions and logs =="
  # keep the newest server version, remove older ones
  ls -1dt "$HOME"/.vscode-server/cli/servers/Stable-* 2>/dev/null | tail -n +2 | xargs -r rm -rf
  ls -1dt "$HOME"/.vscode-server/bin/* 2>/dev/null | tail -n +2 | xargs -r rm -rf
  rm -rf "$HOME"/.vscode-server/data/logs/* 2>/dev/null || true
  rm -rf "$HOME"/.vscode-server/.cli.*.log 2>/dev/null || true
fi

echo
echo "Done. Remaining:"
mine | cut -c1-160 || true
echo
uptime

# Did the journal survive? (needs /var/log/journal to exist)
journalctl --list-boots
ls -ld /var/log/journal

# OOM kills, hangs, NFS stalls in the previous boot
journalctl -b -1 -k --no-pager | grep -iE 'out of memory|oom|killed process|hung task|blocked for more than|soft lockup|rcu.*stall|nfs.*(not responding|timed out)'

# Last 200 warnings or worse before it went down
journalctl -b -1 -p warning --no-pager | tail -200

# Reboot/shutdown history, and who was logged in
last -x reboot shutdown | head
journalctl -b -1 -u sshd --no-pager | grep -E 'Accepted|session opened|session closed' | tail -50

# If there's no persistent journal, fall back to the classic logs
sudo grep -iE 'oom|killed process|hung task|nfs' /var/log/messages* | tail -50




sar -q -f /var/log/sa/sa23     # load average, run queue, total tasks
sar -r -f /var/log/sa/sa23     # memory
sar -B -f /var/log/sa/sa23     # paging and kswapd activity
sar -u -f /var/log/sa/sa23     # CPU user/system split






# Process and thread counts per user (Tasks was 3139 earlier)
ps -eo user= | sort | uniq -c | sort -rn | head
ps -eLo user= | sort | uniq -c | sort -rn | head

# Memory per user, in MB
ps -eo user=,rss= | awk '{m[$1]+=$2} END {for (u in m) printf "%8.0f MB  %s\n", m[u]/1024, u}' | sort -rn | head

# Top memory processes, and top thread counts
ps -eo pid,user,rss,nlwp,pcpu,etime,args --sort=-rss | cut -c1-170 | head -25
ps -eo pid,user,nlwp,args --sort=-nlwp | cut -c1-170 | head -15

# How many VS Code servers, and whose
ps -eo user=,args= | grep -E 'server-main\.js' | grep -v grep | awk '{print $1}' | sort | uniq -c

# Per-user cgroup view (systemd boxes)
systemd-cgtop -m -n 1 --order=memory | head -20
loginctl list-sessions

# Pressure stall info: shows whether the box is stalling on memory, CPU, or I/O
cat /proc/pressure/memory /proc/pressure/cpu /proc/pressure/io

# Kernel memory hogs (NFS inode and dentry caches are common on EFS)
slabtop -o -s c | head -15

# Live view of reclaim: watch the 'si/so' and 'free' columns

  [ "$n" -gt 0 ] && echo "$n $(stat -c %U $p) $(tr '\0' ' ' < $p/cmdline | cut -c1-80)"
done | sort -rn | head -15
