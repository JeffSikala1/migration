# A. First/last snapshot per disk, days with no snapshot of the /app disk, and the third disk
for v in vol-04e3115aea57b5c70 vol-05ebadfa92a493533 vol-08f7bca75e08f0d3f; do
  echo "== $v"
  aws ec2 describe-snapshots --region us-east-1 --owner-ids self \
    --filters Name=tag:Purpose,Values=atlassian-backup Name=volume-id,Values=$v \
    --query '[sort_by(Snapshots,&StartTime)[0].StartTime, sort_by(Snapshots,&StartTime)[-1].StartTime, Snapshots[0].VolumeSize]' --output text
done

snapdays=$(aws ec2 describe-snapshots --region us-east-1 --owner-ids self \
  --filters Name=tag:Purpose,Values=atlassian-backup Name=volume-id,Values=vol-04e3115aea57b5c70 \
  --query 'Snapshots[].StartTime' --output text | tr '\t' '\n' | cut -c1-10 | sort -u)
d=2026-01-06; while [[ "$d" < "2026-10-11" ]]; do grep -qx "$d" <<< "$snapdays" || echo "no snapshot: $d"; d=$(date -d "$d + 1 day" +%F); done

aws ec2 describe-volumes --region us-east-1 --volume-ids vol-05ebadfa92a493533 --output json
