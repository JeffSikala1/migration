curl -fL https://install-cli.jfrog.io | sh
jf --version



mkdir -p prod_repos_detail
jq -r '.[].key' prod_repos.json | while IFS= read -r key; do
  curl -s -H "Authorization: Bearer $artifactory_admin_api_key_secre" \
    "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/repositories/${key}" \
    -o "prod_repos_detail/${key}.json"
done



# Pass 1: LOCAL repos
jq -r '.[] | select(.type=="LOCAL") | .key' prod_repos.json | while IFS= read -r key; do
  echo "=== creating local: $key ==="
  curl --fail -s -u admin:<DEV_ADMIN_TOKEN> \
    -H "Content-Type: application/json" \
    -X PUT "http://localhost:8082/artifactory/api/repositories/${key}" \
    -d @"prod_repos_detail/${key}.json" || echo "FAILED: $key"
done

# Pass 2: REMOTE repos
jq -r '.[] | select(.type=="REMOTE") | .key' prod_repos.json | while IFS= read -r key; do
  echo "=== creating remote: $key ==="
  curl --fail -s -u admin:<DEV_ADMIN_TOKEN> \
    -H "Content-Type: application/json" \
    -X PUT "http://localhost:8082/artifactory/api/repositories/${key}" \
    -d @"prod_repos_detail/${key}.json" || echo "FAILED: $key"
done

# Pass 3: VIRTUAL repos (member repos must exist by now)
jq -r '.[] | select(.type=="VIRTUAL") | .key' prod_repos.json | while IFS= read -r key; do
  echo "=== creating virtual: $key ==="
  curl --fail -s -u admin:<DEV_ADMIN_TOKEN> \
    -H "Content-Type: application/json" \
    -X PUT "http://localhost:8082/artifactory/api/repositories/${key}" \
    -d @"prod_repos_detail/${key}.json" || echo "FAILED: $key"
done


curl -s -u admin:<DEV_ADMIN_TOKEN> "http://localhost:8082/artifactory/api/repositories" | jq 'length'
