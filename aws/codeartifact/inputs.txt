docker exec artifactory bash -c '
for repo in conexus-release-local conexus-snapshot-local; do
  live=$(grep -c "/artifactory/${repo}/" /opt/jfrog/artifactory/var/log/artifactory-request.log 2>/dev/null)
  archived=$(zgrep -h "/artifactory/${repo}/" /opt/jfrog/artifactory/var/log/archived/artifactory-request.*.log.gz 2>/dev/null | wc -l)
  echo "$repo: live=${live:-0} archived=$archived"
done
'



docker exec artifactory bash -c 'head -3 /opt/jfrog/artifactory/var/log/artifactory-request.log'
