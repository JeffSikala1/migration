#!/bin/sh
echo "=== Post-build cnxs-pg-data inspection ==="
echo "Working directory: $(pwd)"
echo ""
echo "--- integration/target/ contents ---"
ls -la integration/target/ 2>&1 || echo "integration/target missing"
echo ""
echo "--- integration/target/cnxs-pg-data contents ---"
ls -la integration/target/cnxs-pg-data/ 2>&1 || echo "cnxs-pg-data missing"
echo ""
echo "--- find all files in cnxs-pg-data ---"
find integration/target/cnxs-pg-data -type f -ls 2>&1 || echo "find returned nothing"
echo ""
echo "--- find any pg-dumpall.sql anywhere ---"
find . -name "full-pg-dumpall.sql" -type f 2>&1 || echo "no pg-dumpall.sql found"
echo "=== End of inspection ==="
