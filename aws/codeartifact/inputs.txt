# List every repo key so we know the real names
curl -s -H "Authorization: Bearer $ARTIFACTORY_TOKEN" \
  "https://<your-artifactory-host>/artifactory/api/repositories" \
  | python3 -m json.tool


curl -s -X POST -H "Authorization: Bearer $ARTIFACTORY_TOKEN" \
  -H "Content-Type: text/plain" \
  "https://<your-artifactory-host>/artifactory/api/search/aql" \
  -d 'items.find({"repo":"cnxs-snapshot-local"})
      .include("repo","path","name","created","size")
      .sort({"$desc":["created"]})
      .limit(500)' \
  | python3 -m json.tool > snapshot_local_results.json

wc -l snapshot_local_results.json

  -H "Content-Type: text/plain" \
  "https://<your-artifactory-host>/artifactory/api/search/aql" \
  -d 'items.find({"repo":"cnxs-snapshot-local"})
      .include("repo","name","created","stat.downloads","stat.downloaded")
      .sort({"$desc":["stat.downloaded"]})
      .limit(200)' \
  | python3 -m json.tool
