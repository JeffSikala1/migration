# repo detail
mkdir -p prod_repos_detail
jq -r '.[].key' prod_repos.json | while IFS= read -r key; do
  curl -s -H "Authorization: Bearer $artifactory_admin_api_key_secret" \
    "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/repositories/${key}" \
    -o "prod_repos_detail/${key}.json"
done
echo "repo details: $(ls prod_repos_detail | wc -l)"

# group detail
mkdir -p prod_groups_detail
jq -r '.[].name' prod_groups.json | while IFS= read -r name; do
  curl -s -H "Authorization: Bearer $artifactory_admin_api_key_secret" \
    "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/security/groups/${name}" \
    -o "prod_groups_detail/${name}.json"
done
echo "group details: $(ls prod_groups_detail | wc -l)"

# permission target detail
mkdir -p prod_permissions_detail
jq -r '.[].name' prod_permissions.json | while IFS= read -r name; do
  encoded=$(python3 -c "import urllib.parse,sys; print(urllib.parse.quote(sys.argv[1]))" "$name")
  curl -s -H "Authorization: Bearer $artifactory_admin_api_key_secret" \
    "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/v2/security/permissions/${encoded}" \
    -o "prod_permissions_detail/${name// /_}.json"
done
echo "permission details: $(ls prod_permissions_detail | wc -l)"
