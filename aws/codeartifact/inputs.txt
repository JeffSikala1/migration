#!/usr/bin/env bash
# ============================================================================
# Restore Bitbucket + Bamboo from prod (.34) backups onto dev (.233),
# alongside the existing dev-artifactory / dev-nginx / dev-postgres stack.
#
# Run this section-by-section, not as one blind execution. Several sections
# need a value filled in (marked TODO) before they'll work, and a few steps
# are genuinely destructive if run twice carelessly (DB creates, container
# names) - read each block before running it.
#
# Assumes: run as root on ip-10-56-128-233 (.233), AFTER the instance has
# already been resized to t3.4xlarge and is back up.
# ============================================================================
set -euo pipefail

# ---------------------------------------------------------------------------
# 0) Pick the backup date to restore from. List what's in S3 first.
# ---------------------------------------------------------------------------
aws s3 ls s3://cnxs-atlassian-backups/apps/
aws s3 ls s3://cnxs-atlassian-backups/postgres/

# TODO: set this to the date folder you want, e.g. 2026-08-19
RESTORE_DATE="TODO-YYYY-MM-DD"

# ---------------------------------------------------------------------------
# 1) Create bind-mount directories matching dev's existing per-service
#    convention (dev uses /app/<service> bind mounts, not named volumes -
#    matching dev-artifactory's /app/jfrog, dev-postgres's /app/postgres).
# ---------------------------------------------------------------------------
mkdir -p /app/bitbucket /app/bamboo /app/restore-staging

# ---------------------------------------------------------------------------
# 2) Pull the app tarballs and postgres dumps from S3 for the chosen date.
# ---------------------------------------------------------------------------
aws s3 cp "s3://cnxs-atlassian-backups/apps/${RESTORE_DATE}/" /app/restore-staging/apps/ --recursive
aws s3 cp "s3://cnxs-atlassian-backups/postgres/${RESTORE_DATE}/" /app/restore-staging/postgres/ --recursive

ls -lh /app/restore-staging/apps/
ls -lh /app/restore-staging/postgres/

# TODO: fill in the exact filenames from the ls output above (they include a
# timestamp suffix, e.g. bitbucket_20260819_021500.tar.gz)
BITBUCKET_TAR="/app/restore-staging/apps/TODO_bitbucket_*.tar.gz"
BAMBOO_TAR="/app/restore-staging/apps/TODO_bamboo_*.tar.gz"
BITBUCKET_DUMP="/app/restore-staging/postgres/TODO_bitbucket_*.dump"
BAMBOO_DUMP="/app/restore-staging/postgres/TODO_bamboo_*.dump"

# ---------------------------------------------------------------------------
# 3) Extract app data into the new bind-mount dirs.
#    The tarballs contain the full path
#    var/lib/docker/volumes/{bitbucketVolume,optbitbucketVolume}/_data/...
#    (that's how they were tar'd on .34) - strip that prefix so the content
#    lands directly under /app/bitbucket and /app/bamboo.
# ---------------------------------------------------------------------------
tar -xzf $BITBUCKET_TAR -C /app/bitbucket --strip-components=5 \
  var/lib/docker/volumes/bitbucketVolume/_data
tar -xzf $BAMBOO_TAR -C /app/bamboo --strip-components=5 \
  var/lib/docker/volumes/bambooVolume/_data

# NOTE: the tarball also has optbitbucketVolume/optbambooVolume (the /opt
# install dirs). We are deliberately NOT restoring those - a fresh image
# provides /opt/atlassian cleanly, and mixing an old install dir with a
# fresh image version has caused upgrade weirdness before. Only the
# application-data (BITBUCKET_HOME / BAMBOO_HOME) content matters here.

ls -la /app/bitbucket
ls -la /app/bamboo

# ---------------------------------------------------------------------------
# 4) Postgres restore.
#    IMPORTANT: dev-postgres was initialized with POSTGRES_USER=artifactory
#    (not the default "postgres" superuser prod's dumps were taken as). So:
#      - do NOT apply prod's globals_*.sql blindly (it references role
#        "postgres" which likely doesn't exist here as a distinct role)
#      - create bitbucket/bamboo DBs owned by the existing "artifactory"
#        superuser role
#      - restore with --no-owner --no-privileges so pg_restore doesn't choke
#        trying to ALTER OWNER / GRANT to prod role names that don't exist
#        in dev-postgres
# ---------------------------------------------------------------------------

# sanity check what roles actually exist in dev-postgres first
docker exec dev-postgres psql -U artifactory -d artifactory -c "\du"

docker exec dev-postgres psql -U artifactory -d artifactory \
  -c "CREATE DATABASE bitbucket OWNER artifactory;"
docker exec dev-postgres psql -U artifactory -d artifactory \
  -c "CREATE DATABASE bamboo OWNER artifactory;"

docker exec -i dev-postgres pg_restore -U artifactory -d bitbucket \
  --no-owner --no-privileges < $BITBUCKET_DUMP

docker exec -i dev-postgres pg_restore -U artifactory -d bamboo \
  --no-owner --no-privileges < $BAMBOO_DUMP

# ---------------------------------------------------------------------------
# 5) CRITICAL: fix the baked-in DB connection config in the restored app
#    data. Prod's Bitbucket/Bamboo point at prod-postgres with prod's
#    creds/hostname - that's now sitting inside the files you just
#    extracted. These MUST be updated before the containers will start
#    cleanly against dev-postgres.
#
#    Bitbucket: /app/bitbucket/shared/bitbucket.properties
#      look for jdbc.url / jdbc.user / jdbc.password (or a <jdbc> block)
#    Bamboo:    /app/bamboo/bamboo.cfg.xml
#      look for hibernate.connection.url / .username / .password
#
#    Point both at: jdbc:postgresql://dev-postgres:5432/bitbucket (or
#    /bamboo), user=artifactory, password=<dev-postgres's actual password -
#    check the dev-postgres container env: POSTGRES_PASSWORD>
# ---------------------------------------------------------------------------
grep -n "jdbc\|postgres" /app/bitbucket/shared/bitbucket.properties || true
grep -n "hibernate.connection" /app/bamboo/bamboo.cfg.xml || true

echo ">>> STOP HERE and manually edit the two files above before continuing <<<"
echo ">>> Confirm dev-postgres's actual password: docker exec dev-postgres env | grep POSTGRES_PASSWORD"

# ---------------------------------------------------------------------------
# 6) Stand up the containers, mirroring prod's config but with dev hostnames
#    in ATL_PROXY_NAME (do NOT reuse the prod hostnames here - see chat notes
#    on why: it changes what URLs these apps generate for themselves).
# ---------------------------------------------------------------------------
docker run -d \
  --name dev-bitbucket \
  --network dev-net \
  --restart unless-stopped \
  -p 7990:7990 \
  -p 7999:7999 \
  -v /app/bitbucket:/var/atlassian/application-data/bitbucket \
  -e ATL_PROXY_NAME=bitbucket-test.dev.cnxs.vpcaas.fcs.gsa.gov \
  -e ATL_PROXY_PORT=443 \
  -e ATL_TOMCAT_SCHEME=https \
  -e ATL_TOMCAT_SECURE=true \
  -e JVM_MINIMUM_MEMORY=1g \
  -e JVM_MAXIMUM_MEMORY=2g \
  atlassian/bitbucket:9.4.17

docker run -d \
  --name dev-bamboo \
  --network dev-net \
  --restart unless-stopped \
  -p 8085:8085 \
  -p 8443:8443 \
  -v /app/bamboo:/var/atlassian/application-data/bamboo \
  -e ATL_PROXY_NAME=bamboo-test.dev.cnxs.vpcaas.fcs.gsa.gov \
  -e ATL_PROXY_PORT=443 \
  -e ATL_TOMCAT_SCHEME=https \
  atlassian/bamboo:10.2.11

# ---------------------------------------------------------------------------
# 7) nginx vhost for Bitbucket (Bamboo doesn't need one - prod exposes it
#    directly on 8085/8443, no nginx in front, per cnxs-mgmt-bitbucket
#    pattern being the ONLY nginx-fronted app on .34).
#    Uses dev's EXISTING cert (do not use the mgmt_wildcard cert from prod's
#    nginx backup - wrong domain, *.mgmt not *.dev).
# ---------------------------------------------------------------------------
cat > /app/nginx/conf.d/bitbucket-test.dev.conf << 'EOF'
server {
    server_name bitbucket-test.dev.cnxs.vpcaas.fcs.gsa.gov;
    listen 80;
    return 301 https://$http_host/;
}

server {
    listen 443 ssl;
    http2 on;
    listen [::]:443 ssl;
    server_name bitbucket-test.dev.cnxs.vpcaas.fcs.gsa.gov;
    ssl_certificate /etc/nginx/certificates/CHANGE_ME_dev_cert.pem;
    ssl_certificate_key /etc/nginx/certificates/CHANGE_ME_dev_key.key;
    resolver 127.0.0.11 ipv6=off valid=30s;

    location / {
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        proxy_set_header X-Forwarded-Proto https;
        proxy_pass http://dev-bitbucket:7990;
        proxy_http_version 1.1;

        client_max_body_size 0;
        proxy_read_timeout  600s;
        proxy_send_timeout  600s;
        send_timeout        600s;
    }
}
EOF

# TODO: replace CHANGE_ME_dev_cert.pem / CHANGE_ME_dev_key.key with the
# actual filenames already in /app/nginx/certificates on .233 (whatever
# dev-nginx currently uses for artifactory-test.dev...) - check with:
ls /app/nginx/certificates/

docker exec dev-nginx nginx -t
docker restart dev-nginx

echo "==== Done with local config. Remaining, NOT scripted here: ===="
echo " - DNS + NLB listener/target-group work for bitbucket-test.dev... (and"
echo "   bamboo-test.dev... if you want it externally reachable too) - same"
echo "   process as the Artifactory NLB work."
echo " - Log into Bitbucket and Bamboo admin UI once up and reset their"
echo "   configured Base URL to the new dev-test hostnames (prod's value is"
echo "   restored inside the app data/DB and won't self-correct)."
