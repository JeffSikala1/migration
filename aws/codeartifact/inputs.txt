  https://artifactory-test.dev.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/system/configuration \

# nginx vhosts backup

# Bamboo: any bandana values containing "-test" hostnames
docker exec dev-postgres psql -U bamboo -d bamboo -c \
  "SELECT plugin_key, bandana_key, bandana_value FROM bandana WHERE bandana_value LIKE '%bamboo-test%' OR bandana_value LIKE '%bitbucket-test%' OR bandana_value LIKE '%artifactory-test%';"

# Bitbucket: any app_property rows containing "-test" hostnames
docker exec dev-postgres psql -U bitbucket -d bitbucket -c \
  "SELECT * FROM app_property WHERE property_value LIKE '%bitbucket-test%' OR property_value LIKE '%bamboo-test%' OR property_value LIKE '%artifactory-test%';"

# Artifactory: current customBaseUrl / proxy settings
