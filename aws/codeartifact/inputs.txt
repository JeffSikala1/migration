# what networks/aliases does the nginx container actually have right now?
docker inspect dev-nginx --format '{{json .NetworkSettings.Networks}}' | python3 -m json.tool

# does DNS resolution inside bamboo/bitbucket actually point at nginx, or go external?
docker exec dev-bamboo getent hosts bitbucket.dev.cnxs.vpcaas.fcs.gsa.gov
docker exec dev-bamboo getent hosts bamboo.dev.cnxs.vpcaas.fcs.gsa.gov
docker exec dev-bitbucket getent hosts bamboo.dev.cnxs.vpcaas.fcs.gsa.gov
docker exec dev-bitbucket getent hosts bitbucket.dev.cnxs.vpcaas.fcs.gsa.gov

# and does a direct curl from inside bamboo to bitbucket's public hostname actually work?
docker exec dev-bamboo curl -sk -o /dev/null -w "%{http_code}\n" --max-time 10 https://bitbucket.dev.cnxs.vpcaas.fcs.gsa.gov
docker exec dev-bitbucket curl -sk -o /dev/null -w "%{http_code}\n" --max-time 10 https://bamboo.dev.cnxs.vpcaas.fcs.gsa.gov
