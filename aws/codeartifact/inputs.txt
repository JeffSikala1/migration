sudo bash -c '
source /app/bbmirror/etc/mirror.env
R=temp-application-01
export GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=http.extraHeader
export GIT_TERMINAL_PROMPT=0
export GIT_CONFIG_VALUE_0="Authorization: Bearer $PROD_TOKEN"
git ls-remote --heads --tags "$PROD_URL/scm/cnxs/$R.git" | sort > /tmp/prod_refs.txt
export GIT_CONFIG_VALUE_0="Authorization: Bearer $DEV_TOKEN"
git ls-remote --heads --tags "$DEV_URL/scm/cnxs/$R.git" | sort > /tmp/dev_refs.txt
wc -l /tmp/prod_refs.txt /tmp/dev_refs.txt
diff /tmp/prod_refs.txt /tmp/dev_refs.txt && echo "REFS MATCH"
'
