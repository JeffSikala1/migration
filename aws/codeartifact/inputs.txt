sudo bash -c '
source /app/bbmirror/etc/mirror.env
export GIT_TERMINAL_PROMPT=0 GIT_CONFIG_COUNT=1 GIT_CONFIG_KEY_0=http.extraHeader
bad=0; n=0
for gd in /app/bbmirror/cache/CNXS/*.git; do
  R=$(basename "$gd" .git); n=$((n+1))
  export GIT_CONFIG_VALUE_0="Authorization: Bearer $PROD_TOKEN"
  p=$(git ls-remote --heads --tags "$PROD_URL/scm/cnxs/$R.git" | sort | md5sum)
  export GIT_CONFIG_VALUE_0="Authorization: Bearer $DEV_TOKEN"
  d=$(git ls-remote --heads --tags "$DEV_URL/scm/cnxs/$R.git" | sort | md5sum)
  [ "$p" = "$d" ] || { echo "MISMATCH: $R"; bad=$((bad+1)); }
done
echo "checked=$n mismatches=$bad"
'
