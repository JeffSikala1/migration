sudo tee /usr/local/sbin/dev-block-prod.sh >/dev/null <<'EOF'
#!/bin/bash
# Dev guardrail: reject container traffic to the prod mgmt NLB. Host traffic is unaffected.
set -eu
BLOCK="10.56.119.26 10.56.119.8"
RULE="-m comment --comment dev-block-prod -j REJECT"
case "${1:-start}" in
  start) for ip in $BLOCK; do iptables -C DOCKER-USER -d "$ip" $RULE 2>/dev/null || iptables -I DOCKER-USER 1 -d "$ip" $RULE; done ;;
  stop)  for ip in $BLOCK; do iptables -D DOCKER-USER -d "$ip" $RULE 2>/dev/null || true; done ;;
esac
EOF
sudo chmod 755 /usr/local/sbin/dev-block-prod.sh

sudo tee /etc/systemd/system/dev-block-prod.service >/dev/null <<'EOF'
[Unit]
Description=Block container traffic to prod mgmt NLB (dev guardrail)
After=docker.service
Requires=docker.service
PartOf=docker.service

[Service]
Type=oneshot
RemainAfterExit=yes
ExecStart=/usr/local/sbin/dev-block-prod.sh start
ExecStop=/usr/local/sbin/dev-block-prod.sh stop

[Install]
WantedBy=multi-user.target
EOF
sudo systemctl daemon-reload && sudo systemctl enable --now dev-block-prod.service
