# List permission targets that grant the anonymous user read
curl -sf -u admin:LEGACY_ADMIN_TOKEN \
  "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions" \
  | jq -r '.[].name' \
  | while read p; do
      result=$(curl -sf -u admin:LEGACY_ADMIN_TOKEN \
        "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions/${p}" \
        | jq -r 'select(.repo.actions.users.anonymous != null) | "\(.name): \(.repo.repositories | join(","))"')
      [ -n "$result" ] && echo "$result"
    done
