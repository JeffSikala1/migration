for c in ip-10-56-112-35.ec2.internal-ba-{1..6}; do
  echo "=== $c ==="
  docker exec "$c" bash -lc '
    echo "PID 1 umask:"
    grep -i Umask /proc/1/status

    echo "Bamboo/Java process umask:"
    p=$(pgrep -f "bamboo|java" | head -1)
    [ -n "$p" ] && grep -i Umask /proc/$p/status || echo "No Bamboo/Java process found"
  '
done
