for p in $(curl -sf -u admin:${LEGACY_TOKEN} \
  "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions" \
  | jq -r '.[].name'); do

  curl -sf -u admin:${LEGACY_TOKEN} \
    "https://conexus-artifactory.edc.ds1.usda.gov/artifactory/api/v2/security/permissions/${p}" \
  | jq -r --arg name "$p" '
      .repo as $repo
      | if ($repo.actions.users // {}) | keys | length > 0 then
          "[\($name)] USERS: \($repo.actions.users)  REPOS: \($repo.repositories | join(","))"
        else empty end,
        if ($repo.actions.groups // {}) | keys | length > 0 then
          "[\($name)] GROUPS: \($repo.actions.groups) REPOS: \($repo.repositories | join(","))"
        else empty end
    '
done
