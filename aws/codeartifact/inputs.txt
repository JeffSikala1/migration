# Pull the raw license.string value and check for embedded whitespace/newlines that would break base64 decoding
docker exec prod-bamboo grep -A1 'license.string' /var/atlassian/application-data/bamboo/bamboo.cfg.xml

# Check byte length and look for anything non-base64 (spaces, newlines, XML entities)
docker exec prod-bamboo sh -c "grep -oP '(?<=license.string\">).*(?=</property>)' /var/atlassian/application-data/bamboo/bamboo.cfg.xml | wc -c"
docker exec prod-bamboo sh -c "grep -oP '(?<=license.string\">).*(?=</property>)' /var/atlassian/application-data/bamboo/bamboo.cfg.xml | od -c | grep -E ' \\\\n| +' | head -5"


docker exec prod-bamboo grep -i 'license.string\|serverId\|serverKey' /var/atlassian/application-data/bamboo/bamboo.cfg.xml
