#!/usr/bin/env bash
set -euo pipefail
source ~/.bashrc

export NODE_EXTRA_CA_CERTS=/etc/pki/ca-trust/source/anchors/USDA_Enterprise_Root_CA.crt

# Install/Update Google Chrome for Headless testing
curl https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/file-local/google-chrome.repo > /etc/yum.repos.d/google-chrome.repo
yum install -y google-chrome-stable

# Configure npm to use Artifactory with auth
cat > ~/.npmrc <<EOF
registry=https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/npm/npm/
//artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory/api/npm/npm/:_authToken=${bamboo.artifactory_access_token_secret}
always-auth=true
EOF

npm install
npm test

ng build --configuration production --output-hashing=all --optimization --source-map=true
