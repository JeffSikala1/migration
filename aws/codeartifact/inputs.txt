for i in 1 2 3 4 5 6; do
  for CACHE_DIR in /cache/bamboo/ip-10-56-112-35.ec2.internal-ba-${i}/xml-data/build-dir/_git-repositories-cache/*/; do
    [ -d "$CACHE_DIR" ] || continue
    # Only target caches that have billingsftp stale release refs
    if git -C "$CACHE_DIR" for-each-ref --format='%(refname)' refs/remotes/origin/release/ 2>/dev/null | grep -q .; then
      echo "=== Pruning $CACHE_DIR ==="
      git -C "$CACHE_DIR" for-each-ref --format='%(refname)' refs/remotes/origin/release/ 2>/dev/null
      GIT_SSH_COMMAND='ssh -i /opt/bamboo-keys/bitbucket_ci -o IdentitiesOnly=yes' \
      git -C "$CACHE_DIR" fetch --prune \
        ssh://git@bitbucket.mgmt.cnxs.vpcaas.fcs.gsa.gov:7999/cnxs/billingsftp.git \
        '+refs/heads/*:refs/remotes/origin/*' 2>/dev/null || true
      echo "After prune:"
      git -C "$CACHE_DIR" for-each-ref --format='%(refname)' refs/remotes/origin/release/ 2>/dev/null || echo "(none)"
    fi
  done
done
