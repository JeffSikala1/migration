# Run on ip-10-56-112-35

# Set up vars (matching the script's defaults)
NEW_IMAGE="339713019047.dkr.ecr.us-east-1.amazonaws.com/build-agents:node22-angular20-el9-2026-05-16-1740"
HOSTNAME_SHORT="$(hostname)"
AGENT_NAME="${HOSTNAME_SHORT}-ba-1"
HOST_AGENT_HOME="/cache/bamboo/${AGENT_NAME}"
CONTAINER_AGENT_HOME="/cache/bamboo/${AGENT_NAME}"
BAMBOO_URL="https://bamboo.mgmt.cnxs.vpcaas.fcs.gsa.gov/agentServer/"
NETWORK_NAME="isolated_nw"
DOCKER_SOCK="/var/run/docker.sock"
SSH_KEYS_DIR="/opt/bamboo-keys"

# Pull the new image first so the swap window is small
docker pull "$NEW_IMAGE"

# (Bamboo UI: disable ba-1 BEFORE proceeding past this point)

# Stop and remove only ba-1
docker rm -f "$AGENT_NAME"

# Start ba-1 with the new image — exact same docker run shape as the script
mkdir -p "$HOST_AGENT_HOME"
docker run -d \
  --net="$NETWORK_NAME" \
  -v "${HOST_AGENT_HOME}:${CONTAINER_AGENT_HOME}" \
  -v "${DOCKER_SOCK}:${DOCKER_SOCK}" \
  -v "${SSH_KEYS_DIR}:/opt/bamboo-keys:ro" \
  --name="$AGENT_NAME" \
  -h "$AGENT_NAME" \
  --init \
  --restart=always \
  --sysctl net.ipv4.ip_forward=1 \
  "$NEW_IMAGE" \
  bash -lc "
    set -e
    echo 'startup umask before change:' \$(umask)
    umask 0002
    echo 'startup umask after change:' \$(umask)
    mkdir -p /opt/java
    ln -sfn /etc/alternatives/java_sdk_17 /opt/java/openjdk
    mkdir -p /opt/bamboo-keys
    mkdir -p '${CONTAINER_AGENT_HOME}'
    exec java -Dbamboo.home='${CONTAINER_AGENT_HOME}' -jar /opt/bamboo-agent.jar '${BAMBOO_URL}'
  "

# Verify it's running
docker ps --filter "name=${AGENT_NAME}" --format "table {{.Names}}\t{{.Image}}\t{{.Status}}"

# Watch the agent connect to Bamboo
docker logs -f "$AGENT_NAME"
