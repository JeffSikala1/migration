import os, requests

PROD_URL = "https://artifactory.mgmt.cnxs.vpcaas.fcs.gsa.gov/artifactory"
DEV_URL = "https://artifactory-test.dev.cnxs.vpcaas.fcs.gsa.gov/artifactory"
PROD_TOKEN = os.environ["artifactory_prod_admin_token"]
DEV_TOKEN = os.environ["artifactory_dev_admin_token"]

def list_paths(base_url, token, repo):
    resp = requests.post(
        f"{base_url}/api/search/aql",
        headers={"Authorization": f"Bearer {token}", "Content-Type": "text/plain"},
        data=f'items.find({{"repo":"{repo}","type":"file"}}).include("path","name")',
        timeout=120,
    )
    resp.raise_for_status()
    return {f'{i["path"]}/{i["name"]}' for i in resp.json()["results"]}

repo = "conexus-snapshot-local"
prod = list_paths(PROD_URL, PROD_TOKEN, repo)
dev = list_paths(DEV_URL, DEV_TOKEN, repo)

missing = prod - dev
print(f"prod: {len(prod)}, dev: {len(dev)}, missing from dev: {len(missing)}")
for p in sorted(missing):
    print(f"  MISSING: {p}")

extra = dev - prod
if extra:
    print(f"unexpected extra in dev: {len(extra)}")
    for p in sorted(extra):
        print(f"  EXTRA: {p}")
EOF
