sudo bash -c 'source /app/bbmirror/etc/oneshot.env
start=0
while :; do
  r=$(curl -sS --fail -K - "$PROD_URL/rest/api/1.0/repos?limit=100&start=$start" <<<"header = \"Authorization: Bearer $PROD_TOKEN\"")
  jq -r ".values[] | [.project.key,.project.type,.state] | @tsv" <<<"$r"
  [ "$(jq -r .isLastPage <<<"$r")" = true ] && break
  start=$(jq -r .nextPageStart <<<"$r")
done' | sort | uniq -c
