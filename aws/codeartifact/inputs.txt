sudo bash -c '
R=contract-mod
gd=/app/bbmirror/cache/CNXS/$R.git
for ref in refs/heads/develop refs/heads/feature/ll-modelchanges; do
  d=$(grep "^$ref " /tmp/dev_$R.txt | cut -d" " -f2)
  p=$(grep "^$ref " /tmp/prod_$R.txt | cut -d" " -f2)
  if git --git-dir=$gd cat-file -e "$d^{commit}" 2>/dev/null; then
    git --git-dir=$gd merge-base --is-ancestor "$d" "$p" && echo "$ref: FAST-FORWARD (dev is behind prod)" || echo "$ref: DIVERGED (dev has commits prod lacks)"
  else
    echo "$ref: DIVERGED (dev commit $d not in prod history)"
  fi
done
'


sudo -v
sudo nohup /app/bbmirror/bin/cnxs_mirror.sh > /tmp/cnxs_full3.log 2>&1 &
tail -f /tmp/cnxs_full3.log
