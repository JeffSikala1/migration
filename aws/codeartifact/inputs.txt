for c in ip-10-56-112-35.ec2.internal-ba-{1..6}; do
  echo "=== $c ==="
  docker exec "$c" bash -lc '
    echo "Container startup logs:"
    echo

    echo "Process tree:"
    ps -eo pid,ppid,comm,args | grep -E "[j]ava|[b]amboo-agent|[t]ini|[b]ash"

    echo
    p=$(ps -eo pid,args | awk "/[j]ava .*\/opt\/bamboo-agent.jar/ {print \$1; exit}")
    echo "Selected Bamboo agent PID: ${p:-not found}"

    if [ -n "$p" ]; then
      grep -i Umask /proc/$p/status
    fi
  '
done

done




for c in ip-10-56-112-35.ec2.internal-ba-{1..6}; do
  echo "=== $c ==="
  docker exec "$c" bash -lc '
    ps -eo pid,ppid,comm,args | grep -E "[j]ava|[b]amboo-agent|[w]rapper"

    echo
    echo "=== Umask by process ==="
    for p in $(ps -eo pid,comm,args | awk '\''$2=="java" || $2=="wrapper" || $2=="bamboo-agent.sh" {print $1}'\''); do
      echo "--- PID $p ---"
      ps -p "$p" -o pid,ppid,comm,args
      grep -i Umask /proc/$p/status
    done
  '
done
